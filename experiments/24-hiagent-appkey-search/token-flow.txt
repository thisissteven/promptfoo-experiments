const memLocalCache: Record<string, string> = {};
const memSessionCache: Record<string, string> = {};

let warnedSetFailure = false;
let warnedReadback = false;

function warnSetFailure(key: string, err: unknown): void {
  if (warnedSetFailure) return;
  warnedSetFailure = true;
  console.warn("[track] storage setItem threw", { key, err });
}

function warnReadback(
  key: string,
  expected: string,
  actual: string | null,
): void {
  if (warnedReadback) return;
  warnedReadback = true;
  console.warn("[track] storage setItem readback mismatch", {
    key,
    expected,
    actual,
  });
}

export function safeLocalSet(key: string, value: string): boolean {
  memLocalCache[key] = value;
  if (typeof localStorage === "undefined") return false;
  try {
    localStorage.setItem(key, value);
  } catch (err) {
    warnSetFailure(key, err);
    return false;
  }
  try {
    const actual = localStorage.getItem(key);
    if (actual !== value) {
      warnReadback(key, value, actual);
      return false;
    }
  } catch {
    return false;
  }
  return true;
}

export function safeLocalGet(key: string): string | null {
  if (typeof localStorage !== "undefined") {
    try {
      const v = localStorage.getItem(key);
      if (v !== null) return v;
    } catch {
      // σñ▒Φ┤Ñσ¢₧ΘÇÇσê░σåàσ¡ÿτ╝ôσ¡ÿ
    }
  }
  return Object.prototype.hasOwnProperty.call(memLocalCache, key)
    ? memLocalCache[key]
    : null;
}

function safeSessionSet(key: string, value: string): boolean {
  memSessionCache[key] = value;
  if (typeof sessionStorage === "undefined") return false;
  try {
    sessionStorage.setItem(key, value);
    return true;
  } catch (err) {
    warnSetFailure(key, err);
    return false;
  }
}

function safeSessionGet(key: string): string | null {

}

export function getTempUserId(): string {

const initialUrlChannelSnapshot: { channel: string; from: string } = (() => {
  if (typeof window === "undefined") return { channel: "", from: "" };
  try {
    const params = new URLSearchParams(window.location.search);
    return {
      channel: params.get("channel") || "",
      from: params.get("from") || "",
    };
  } catch {
    return { channel: "", from: "" };
  }
})();

 * τö▒ TrackProvider σ£¿ init Σ╣ïσÉÄΘÇÜΦ┐ç tracker.updateCommonData Φªåτ¢ûπÇé

export function getChannelCode(): string {
  if (typeof window === "undefined") return "";
  if (initialUrlChannelSnapshot.channel)
    return initialUrlChannelSnapshot.channel;
  if (initialUrlChannelSnapshot.from) return initialUrlChannelSnapshot.from;
  const params = new URLSearchParams(window.location.search);
  const fromUrl = params.get("channel") || params.get("from");
  if (fromUrl) return fromUrl;
  // sessionStorage σà£σ║ò∩╝êΣ╕Ä useURLSearchParams.ts τÜä CHANNEL_STORAGE_KEY σ»╣Θ╜É∩╝ë
  // Σ╕ìτ¢┤µÄÑ import Θü┐σàìσÆî hooks µûçΣ╗╢ΦÇªσÉê∩╝îτö¿ try/catch Θÿ▓ storage Σ╕ìσÅ»τö¿
  try {
    const stored = sessionStorage.getItem("qr_channel");
    if (stored) return stored;
  } catch {
    // ignore
  }
  if (isHandyApp()) return `${getChannelLangPrefix()}-handy`;
  return "";
}

function readActiveTabs(): ActiveTabsMap {

    if (!parsed || typeof parsed !== "object") return {};

    for (const [k, v] of Object.entries(parsed)) {
      if (typeof v === "number") out[k] = v;

function pruneActiveTabs(tabs: ActiveTabsMap): ActiveTabsMap {
  const now = Date.now();

  }
  return out;
}

function base64ToBytes(value: string): Uint8Array {
  const normalized = value.replace(/-/g, "+").replace(/_/g, "/");
  const pad = "=".repeat((4 - (normalized.length % 4)) % 4);
  return Uint8Array.from(atob(normalized + pad), (c) => c.charCodeAt(0));
}

/** TS 5.7+ Σ╕ï Uint8Array.buffer Σ╕║ ArrayBufferLike∩╝îWeb Crypto Θ£ÇΦªü ArrayBuffer */
function toArrayBuffer(bytes: Uint8Array): ArrayBuffer {
  return bytes.buffer.slice(
    bytes.byteOffset,
    bytes.byteOffset + bytes.byteLength,
  ) as ArrayBuffer;
}

async function importAesKey(): Promise<CryptoKey> {
  if (!KEY_B64) {
    throw new Error("NEXT_PUBLIC_ENTRY_AES_KEY is not configured");
  }
  return crypto.subtle.importKey(
    "raw",
    toArrayBuffer(base64ToBytes(KEY_B64)),
    "AES-GCM",
    false,
    ["decrypt"],
  );
}

/**
 * Φºúσ»åσàÑσÅúΦ╖│Θô╛Σ╕¡τÜä bblTokenπÇé
 * σ»åµûçµá╝σ╝ÅΣ╕Ä Java BblTokenAesUtils Σ╕ÇΦç┤∩╝ÜBase64URL(iv[12] || ciphertext || tag[16])∩╝î
 * µÿÄµûçΣ╕║ `token|expireAtMs`πÇé
 */
export async function decryptBblToken(packedB64: string): Promise<string> {
  const packed = base64ToBytes(packedB64.trim());
  if (packed.length < 12 + 16) {
    throw new Error("invalid packed token");
  }
  const iv = packed.slice(0, 12);
  const data = packed.slice(12);
  const raw = await crypto.subtle.decrypt(
    { name: "AES-GCM", iv: toArrayBuffer(iv), tagLength: 128 },
    await importAesKey(),
    toArrayBuffer(data),
  );
  const text = new TextDecoder().decode(raw);
  const sep = text.lastIndexOf("|");

const KEY_B64 = (process.env.NEXT_PUBLIC_ENTRY_AES_KEY ?? "").trim();

export class BblTokenExpiredError extends Error {
  constructor() {
    super("bblToken expired");
    this.name = "BblTokenExpiredError";
  }
}

function base64ToBytes(value: string): Uint8Array {
  const normalized = value.replace(/-/g, "+").replace(/_/g, "/");
  const pad = "=".repeat((4 - (normalized.length % 4)) % 4);
  return Uint8Array.from(atob(normalized + pad), (c) => c.charCodeAt(0));
}

/** TS 5.7+ Σ╕ï Uint8Array.buffer Σ╕║ ArrayBufferLike∩╝îWeb Crypto Θ£ÇΦªü ArrayBuffer */
function toArrayBuffer(bytes: Uint8Array): ArrayBuffer {
  return bytes.buffer.slice(
    bytes.byteOffset,
    bytes.byteOffset + bytes.byteLength,
  ) as ArrayBuffer;
}

async function importAesKey(): Promise<CryptoKey> {
  if (!KEY_B64) {
    throw new Error("NEXT_PUBLIC_ENTRY_AES_KEY is not configured");
  }
  return crypto.subtle.importKey(
    "raw",
    toArrayBuffer(base64ToBytes(KEY_B64)),
    "AES-GCM",
    false,
    ["decrypt"],
  );
}

/**
 * Φºúσ»åσàÑσÅúΦ╖│Θô╛Σ╕¡τÜä bblTokenπÇé
 * σ»åµûçµá╝σ╝ÅΣ╕Ä Java BblTokenAesUtils Σ╕ÇΦç┤∩╝ÜBase64URL(iv[12] || ciphertext || tag[16])∩╝î
 * µÿÄµûçΣ╕║ `token|expireAtMs`πÇé
 */
export async function decryptBblToken(packedB64: string): Promise<string> {
  const packed = base64ToBytes(packedB64.trim());
  if (packed.length < 12 + 16) {
    throw new Error("invalid packed token");
  }
  const iv = packed.slice(0, 12);
  const data = packed.slice(12);
  const raw = await crypto.subtle.decrypt(
    { name: "AES-GCM", iv: toArrayBuffer(iv), tagLength: 128 },
    await importAesKey(),
    toArrayBuffer(data),
  );
  const text = new TextDecoder().decode(raw);
  const sep = text.lastIndexOf("|");
  if (sep <= 0 || sep === text.length - 1) {
    throw new Error("invalid payload");
  }
  const token = text.slice(0, sep);
  const expireAt = Number(text.slice(sep + 1));
  if (!token || !Number.isFinite(expireAt)) {
    throw new Error("invalid payload");
  }
  if (Date.now() > expireAt) {
    throw new BblTokenExpiredError();
  }
  return token;
}

    queueMicrotask(() => {
      if (currentSessionIdRef.current) {
        ensureSessionPersisted(currentSessionIdRef.current);
      }

    // Handy σ«┐Σ╕╗σåàπÇüURL µ£¬µîçσ«Ü channel µù╢∩╝îσ╝éµ¡ÑΣ╗ÄµíÑµÄÑΦÄ╖σÅû HMS τ▓╛σîû channel_code
    // init Θÿ╢µ«╡σ╖▓σàêτö¿ cn-handy / en-handy σìáΣ╜ì∩╝îµíÑµÄÑµêÉσèƒσêÖΦªåτ¢ûΣ╕║ cn-{HMS} / en-{HMS}
    // µíÑµÄÑσñ▒Φ┤Ñ / Φ╢àµù╢ / Φ┐öσ¢₧τ⌐║σ¥çΣ┐¥µîüσìáΣ╜ì∩╝îΣ╕ìΘÿ╗σí₧Σ╕╗µ╡üτ¿ï
    if (shouldFetchHandyHms()) {
      requestUserPromptFromApp()
        .then((res) => {
          const hms = res?.hms || res?.hmsList?.[0]?.hms;
          if (hms) {
            tracker.updateCommonData({
              channel_code: buildHandyHmsChannelCode(hms),
            });
          }
        })
        .catch(() => {});
    }

    resetSessionTimeout();

    // --- µ┤╗Φ╖â tab σ┐âΦ╖│ ---
    // µ»Å 10s σåÖΣ╕Çµ¼íΦç¬σ╖▒τÜä lastHeartbeat σê░ track_active_tabs∩╝îΣ┐¥Φ»üΣ╗ìσ£¿ active_tabs
    // Θ¢åσÉêσåàπÇéσà╢Σ╗û tab σà│Θù¡µù╢ decrementTabCount Σ╝ÜΦ»╗Φ┐ÖΣ╗╜Θ¢åσÉêσêñσ«ÜπÇîµÿ»σÉªµ£ÇσÉÄΣ╕ÇΣ╕¬ tabπÇìπÇé
    // ΘÜÉΦùÅ tab τÜä setInterval Σ╝ÜΦó½µ╡ÅΦºêσÖ¿Φèéµ╡üσê░ 1Hz∩╝îΣ╜å 10s Θù┤ΘÜö + 30s Φ┐çµ£ƒΘÿêσÇ╝
    // Σ╣ïΘù┤τòÖµ£ëσààΦ╢│ buffer∩╝îΦèéµ╡üσ£║µÖ»Σ╕ïΣ╣ƒΦâ╜Φç│σ░æµëôσê░Σ╕Çµïì
    if (tabHeartbeatRef.current) clearInterval(tabHeartbeatRef.current);
    tabHeartbeatRef.current = setInterval(
      heartbeatActiveTab,
      TAB_HEARTBEAT_INTERVAL_MS,
    );

    const handleInteraction = () => {
      // πÇîsession σñìµ┤╗πÇìσêåµö»Σ╕ìσÅùΦèéµ╡üτ║ªµ¥ƒ∩╝Üτí«Σ┐¥ BFCache µüóσñì / timeout Σ╕èµèÑσÉÄ
      // τö¿µê╖τ¼¼Σ╕ÇΣ╕¬τé╣σç╗Σ║ïΣ╗╢σ░▒Φâ╜µîéσ£¿µû░ session Σ╕è∩╝îΘü┐σàìπÇîself_service_click /
      // transfer_human ΦÉ╜σê░σ╖▓ ended τÜä sessionπÇìτÜäΘù«Θóÿ

        const newSessionId = createNewSession();
        currentSessionIdRef.current = newSessionId;
        tracker.updateBaseData({ session_id: newSessionId });
        trackSessionStart();
        updateLastActive();
        resetSessionTimeout();
        lastInteractionTime = Date.now();
        return;
      }

      const now = Date.now();
      if (now - lastInteractionTime < INTERACTION_THROTTLE_MS) return;
      lastInteractionTime = now;

      updateLastActive();
      resetSessionTimeout();
    };

    const interactionEvents = ["mousedown", "keydown", "touchstart", "scroll"];
    interactionEvents.forEach((evt) => {
      window.addEventListener(evt, handleInteraction, { passive: true });

    // event.persisted === true Φí¿τñ║Σ╗Ä BFCache µüóσñì∩╝êuseEffect Σ╕ìΣ╝ÜΘçìΦ╖æ∩╝ë∩╝î

    const handlePageShow = (event: PageTransitionEvent) => {
      if (!event.persisted) return;
      const existing = getSessionId();
      const sessionInvalid =
        !existing || isSessionExpired() || isSessionEnded(existing);
      if (sessionInvalid) {
        // σà│Θö«Σ┐«σñì∩╝ÜBFCache µ£ƒΘù┤ JS µëºΦíîΦó½σå╗τ╗ô∩╝îsetTimeout Σ╕ìΣ╝ÜµÄ¿Φ┐¢∩╝îpagehide
        // Σ╣ƒΣ╕ìΣ╝Üσ╕ª persisted=false ΦºªσÅæ session_endπÇéτ¢┤µÄÑ createNewSession Σ╝ÜτòÖΣ╕ï
        // Σ╕ÇΣ╕¬µ░╕Φ┐£Σ╕ìΘù¡τÄ»τÜäµùº session∩╝êµò░Σ╗ôΣ╛º session_start σñÜΣ║Ä session_end∩╝î

          // µ¢┤µù⌐τÜäΦ┤ƒσÉæµò░µì«πÇéσ£¿ markSessionEnded / σÉÄΘ¥óτÜä createNewSession σëì

          // getSessionDuration σ╖▓σ«₧τÄ░ lastActive - startTime Φ»¡Σ╣ë∩╝î
          // BFCache Φ╖¿σ░Åµù╢Φ╖¿σñ⌐µù╢Σ╕ìΣ╝ÜΦó½ now µ▒íµƒô
          trackSessionEnd("timeout", oldDuration);

      incrementTabCount();

      if (e.key === "track_last_active" && currentSessionIdRef.current) {

          passive: true,

      window.removeEventListener("pagehide", handlePageHide, true);
      window.removeEventListener("pageshow", handlePageShow);
      window.removeEventListener("storage", handleStorage);
      document.removeEventListener("visibilitychange", handleVisibilityChange);
      if (sessionTimeoutRef.current) {
        clearTimeout(sessionTimeoutRef.current);
      }
      if (tabHeartbeatRef.current) {
        clearInterval(tabHeartbeatRef.current);
        tabHeartbeatRef.current = null;
      }

  }, []);

    if (!initializedRef.current) return;
    tracker.updateBaseData({ uid: user?.cusId || "" });
  }, [user?.cusId]);

  return <>{children}</>;
