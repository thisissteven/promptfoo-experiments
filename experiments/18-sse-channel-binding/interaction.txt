function stripNonSerializableRobotData(

export function serializeChatMessage(
  msg: Message | RobotMessage,
): SerializedChatMessage {
  const base: SerializedChatMessage = {
    id: msg.id,
    clientId: msg.clientId,
    content: msg.content ?? "",
    timestamp: msg.timestamp,
    senderType: msg.senderType,
    type: msg.type,
    headerType: msg.headerType,
    senderName: msg.senderName,
  };
  if ("messageType" in msg) {
    const robotMsg = msg as RobotMessage;
    return {
      ...base,
      messageType: robotMsg.messageType,
      extension: robotMsg.extension,
      streaming: robotMsg.streaming,
      error: robotMsg.error,
      robotData: stripNonSerializableRobotData(robotMsg.robotData),
    };
  }
  return base;
}

/**
 * Σ╗Ä anchor σ£¿ msgList Σ╕¡τÜäΣ╜ìτ╜«σÉæΣ╕èµë╛∩╝îτ¢┤σê░ customer∩╝êMessageSender.USER∩╝ëπÇé
 * Φ┐öσ¢₧ [customer .. anchor] Θù¡σî║Θù┤∩╝¢µë╛Σ╕ìσê░ anchor Φ┐öσ¢₧ []∩╝¢µë╛Σ╕ìσê░ customer σêÖΣ╗ÄσêùΦí¿σñ┤σê░ anchorπÇé
 */
export function collectMsgListSliceUpToCustomer(
  msgList: (Message | RobotMessage)[],
  anchorMessageId: string,
): {
  messages: (Message | RobotMessage)[];
  customerMessageId?: string;
} {
  const anchorIndex = msgList.findIndex((m) => m.id === anchorMessageId);
  if (anchorIndex === -1) {
    return { messages: [] };
  }

  let customerIndex = -1;
  for (let i = anchorIndex; i >= 0; i--) {
    if (isCustomerMessage(msgList[i])) {
      customerIndex = i;
      break;
    }
  }

  const startIndex = customerIndex === -1 ? 0 : customerIndex;
  const slice = msgList.slice(startIndex, anchorIndex + 1);
  const customerMessageId =
    customerIndex >= 0 ? msgList[customerIndex].id : undefined;

  return { messages: slice, customerMessageId };
}

/** Σ╗Ä form_push SSE data Φºúµ₧Éµ£ìσèíτ½»Σ╕ïσÅæτÜä clientId */
export function parseFormPushClientIdFromEventData(
  formPushEventData: string,
): string | undefined {
  const parsed = tryParse(formPushEventData.trim());
  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
    return undefined;
  }
  const clientId = (parsed as Record<string, unknown>).clientId;
  if (typeof clientId !== "string" || !clientId.trim()) {
    return undefined;
  }
  return clientId.trim();
}

/** form_push τ╝ôσ¡ÿ∩╝ÜΣ╗àΘöÜτé╣µ╢êµü»µ¢┐µìó clientIdπÇücontent τ╜«τ⌐║∩╝îextension Σ╕║σÄƒσºï event data */
function applyFormPushCacheOverrides(
  messages: SerializedChatMessage[],
  anchorMessageId: string,
  formPushEventData: string,
): SerializedChatMessage[] {
  const overrideClientId =
    parseFormPushClientIdFromEventData(formPushEventData);

  return messages.map((item) => {
    if (item.id !== anchorMessageId) {
      return item;
    }
    return {
      ...item,
      ...(overrideClientId ? { clientId: overrideClientId } : {}),
      type: MessageType.FormPush,
      content: "",
      extension: formPushEventData,
    };
  });
}

export function buildSellingLogisticsSessionCache(params: {
  anchorMessageId: string;
  sseEvent: string;
  scenePayload: Record<string, unknown>;
  msgList: (Message | RobotMessage)[];
  requestId?: string;
  /** form_push Σ║ïΣ╗╢σÄƒσºï data∩╝îτö¿Σ║ÄΘöÜτé╣ clientId / content / extension */
  formPushEventData?: string;
}): SellingLogisticsSessionCache | null {
  const { messages, customerMessageId } = collectMsgListSliceUpToCustomer(
    params.msgList,
    params.anchorMessageId,
  );
  if (messages.length === 0) {
    return null;
  }

  let serialized = messages.map(serializeChatMessage);
  const eventData = params.formPushEventData?.trim();
  if (eventData) {
    serialized = applyFormPushCacheOverrides(
      serialized,
      params.anchorMessageId,
      eventData,
    );
  }

  return {
    version: 1,
    savedAt: Date.now(),
    anchorMessageId: params.anchorMessageId,
    sseEvent: params.sseEvent,
    scenePayload: params.scenePayload,
    requestId: params.requestId,
    customerMessageId,
    messages: serialized,
  };
}

export function saveSellingLogisticsSessionCache(
  payload: SellingLogisticsSessionCache,
): boolean {
  if (typeof sessionStorage === "undefined") return false;
  try {

      JSON.stringify(payload),
    );
    return true;
  } catch (e) {
    console.warn("[sellingLogisticsSessionCache] save failed:", e);
    return false;
  }
}

const FORM_PUSH_SSE_OMIT_KEYS = new Set([
  "extension",
  "content",
  "type",
  "headerType",
  "senderType",

/**
 * Φºúµ₧É form_push Σ║ïΣ╗╢ data µ«╡ JSONπÇé
 * extension σ¡ÿµò┤σîà SSE data∩╝êσÉ½ aiConversationIdπÇüclientId∩╝ë∩╝îΣ╕Ä session τ╝ôσ¡ÿΘöÜτé╣Σ╕ÇΦç┤∩╝î
 * Σ╛¢ parseSellingLogisticsFormMeta µÅÉΣ║ñ∩╝¢σåàσ▒é formType τö▒ parseSellingLogisticsExtension Φºúµ₧ÉπÇé
 */
export function parseFormPushSseData(data: string): {
  extension: string;
  content: string;
  scenePayload: Record<string, unknown>;
} {
  const fallbackExtension = JSON.stringify({
    formType: FormPushType.SellingLogistics,
    submitted: false,
  });
  if (!data?.trim() || data === "[DONE]") {
    return { extension: fallbackExtension, content: "", scenePayload: {} };
  }
  try {
    const parsed = JSON.parse(data) as Record<string, unknown>;
    const content = typeof parsed.content === "string" ? parsed.content : "";
    const scenePayload: Record<string, unknown> = {};
    for (const [key, value] of Object.entries(parsed)) {
      if (FORM_PUSH_SSE_OMIT_KEYS.has(key)) continue;
      scenePayload[key] = value;
    }
    return { extension: data.trim(), content, scenePayload };
  } catch {
    return {
      extension: fallbackExtension,
      content: "",
      scenePayload: { payload: data },
    };
  }
}

}

function ensureUserSaveContent(content: string): string {

}

  const payload: Record<string, unknown> = {

/** form_push ΦÉ╜σ║ô extension Σ╗àΣ┐¥σ¡ÿσåàσ▒éΦí¿σìòσàâµò░µì«∩╝îΦÇîΘ¥₧ SSE µò┤σîà */
function resolveFormPushExtensionForSave(
  extension?: string,
): string | undefined {
  if (!extension?.trim()) return undefined;
  const parsed = tryParse(extension.trim());
  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
    return extension.trim();
  }
  const outer = parsed as Record<string, unknown>;
  const innerExtension = outer.extension;
  if (typeof innerExtension === "string" && innerExtension.trim()) {
    return innerExtension.trim();
  }
  return extension.trim();
}

function toSaveRobotInfoType(
  item: SerializedChatMessage,
  cache: SellingLogisticsSessionCache,
): string {
  if (
    item.senderType === MessageSender.ROBOT &&
    item.id === cache.anchorMessageId
  ) {
    return MessageType.FormPush;
  }
  return item.type ?? MessageType.Text;
}

/** σ░åσö«Σ╕¡τë⌐µ╡ü session τ╝ôσ¡ÿτëçµ«╡Φ╜¼Σ╕║ robot-message ΦÉ╜σ║ôσàÑσÅé */
export function buildSaveRobotInfoPayloadFromSellingLogisticsCache(
  cache: SellingLogisticsSessionCache,
): SaveRobotInfoPropsType[] {
  return cache.messages.map((item) => {
    const clientId = item.clientId ?? item.id ?? "";
    const isUser = item.senderType === MessageSender.USER;
    const type = toSaveRobotInfoType(item, cache);
    const payload: SaveRobotInfoPropsType = {
      clientId,
      clientTimestamp: item.timestamp,
      content: isUser
        ? ensureUserSaveContent(item.content ?? "")
        : type === MessageType.FormPush
          ? ""
          : buildRobotSaveContent(item),
      senderType: item.senderType,
      type,
    };
    if (type === MessageType.FormPush && item.extension?.trim()) {
      payload.extension = resolveFormPushExtensionForSave(item.extension);
    }
    if (item.extension) {
      const jsonExtension = tryParse(item.extension);
      if (jsonExtension && jsonExtension.aiConversationId) {
        payload.aiConversationId = jsonExtension.aiConversationId;
      }
    }
    return payload;
  });
}

function deserializeToRobotMessage(
  item: SerializedChatMessage,
  cache: SellingLogisticsSessionCache,
): RobotMessage {
  const isAnchor = item.id === cache.anchorMessageId;
  return {
    id: item.id,
    clientId: item.clientId,
    type: isAnchor ? MessageType.FormPush : (item.type ?? MessageType.Text),
    headerType: item.headerType ?? HeaderType.MESSAGE,
    senderType: MessageSender.ROBOT,
    messageType: item.messageType ?? RobotMessageType.ROBOT_SSE_HISTORY,
    content: item.content ?? "",
    timestamp: item.timestamp,
    senderName: item.senderName ?? "µÖ║Φâ╜σè⌐µëï",
    extension: isAnchor
      ? item.extension?.trim() || buildAnchorFormPushExtension(item.extension)
      : item.extension,
    streaming: false,
    error: item.error,
    robotData: isAnchor
      ? { ...(item.robotData ?? {}), ...cache.scenePayload }
      : (item.robotData ?? {}),
  };
}

/**
 * σ░å sessionStorage Σ╕¡τÜäΣ╝ÜΦ»¥τëçµ«╡σåÖσ¢₧ msgList∩╝îσåÖσ«îσÉÄµ╕àΘÖñτ╝ôσ¡ÿπÇé
 * @returns µÿ»σÉªµêÉσèƒµüóσñìΦç│σ░æΣ╕Çµ¥íµ╢êµü»
 */
export function restoreSellingLogisticsSessionToChat(actions: {
  addOutgoingMessage: (msg: Message) => Message;
  setRobotMessage: (msg: RobotMessage) => void;
}): boolean {
  const cache = getSellingLogisticsSessionCache();
  if (!cache?.messages?.length) {
    return false;
  }

  for (const item of cache.messages) {
    if (item.senderType === MessageSender.USER) {
      actions.addOutgoingMessage(deserializeToUserMessage(item));
      continue;
    }
    if (item.senderType === MessageSender.ROBOT) {
      actions.setRobotMessage(deserializeToRobotMessage(item, cache));
    }
  }

  clearSellingLogisticsSessionCache();
  stripFromSellingLogisticsUrlParam();
  return true;
}

function toStringArray(value: unknown): string[] {
  if (!Array.isArray(value)) return [];
  return value.filter((v): v is string => typeof v === "string" && !!v.trim());

  return typeof value === "string" && value.trim() ? value : undefined;

function readIndex(value: unknown): number | undefined {
  if (typeof value === "number" && Number.isInteger(value) && value >= 0) {
    return value;
  }
  if (typeof value === "string" && value.trim()) {
    const n = Number(value);
    if (Number.isInteger(n) && n >= 0) return n;
  }
  return undefined;
}

function parseOptions(value: unknown): InteractionCardOption[] {
  if (!Array.isArray(value)) return [];
  return value.reduce<InteractionCardOption[]>((acc, item) => {
    if (item && typeof item === "object" && !Array.isArray(item)) {
      const record = item as Record<string, unknown>;
      const label = typeof record.label === "string" ? record.label.trim() : "";
      if (label) {
        const image = resolveOptionImage(record);
        acc.push({ label, ...(image ? { image } : {}) });
      }
    } else if (typeof item === "string" && item.trim()) {
      acc.push({ label: item.trim() });
    }
    return acc;
  }, []);
}

function parseInfo(value: unknown): InteractionCardInfo {

    options: parseOptions(record.options),
  };

function unwrapExtension(value: unknown): Record<string, unknown> | null {
  let obj: unknown = value;
  if (typeof value === "string") {

export function parseInteractionCardPayload(
  raw?: string | null,
): InteractionCardPayload | null {
  if (raw == null || typeof raw !== "string" || !raw.trim()) {
    return null;
  }
  if (raw === "[DONE]") {
    return null;
  }
  const parsed = tryParse(raw.trim());
  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
    return null;
  }
  const outer = parsed as Record<string, unknown>;
  // µë┐Φ╜╜σìíτëçµò░µì«τÜäσ»╣Φ▒í∩╝ÜΣ╝ÿσàêΣ┐íσ░üσåà extension∩╝îσà╢µ¼íσñûσ▒éµ£¼Φ║½
  const cardHost = unwrapExtension(outer.extension) ?? outer;

  const rawInfo =
    cardHost.interaction_card_info !== undefined
      ? cardHost.interaction_card_info
      : cardHost;
  const info = parseInfo(rawInfo);

  // clientId / card_type / ΘÇëΣ╕¡µÇüΣ╝ÿσàêσÅûσñûσ▒éΣ┐íσ░ü∩╝îσ¢₧ΘÇÇσìíτëçµë┐Φ╜╜σ»╣Φ▒í
  return {
    clientId: readString(outer.clientId) ?? readString(cardHost.clientId),
    cardType:
      info.cardType ??
      readString(outer.card_type) ??
      readString(cardHost.card_type),
    info,
    selectedLabel:
      readString(outer.selectedLabel) ?? readString(cardHost.selectedLabel),
    selectedIndex:
      readIndex(outer.selectedIndex) ?? readIndex(cardHost.selectedIndex),
  };
}
