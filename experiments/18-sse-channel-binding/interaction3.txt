function toStringArray(value: unknown): string[] {
  if (!Array.isArray(value)) return [];
  return value.filter((v): v is string => typeof v === "string" && !!v.trim());
}

function readString(value: unknown): string | undefined {
  return typeof value === "string" && value.trim() ? value : undefined;
}

function readIndex(value: unknown): number | undefined {
  if (typeof value === "number" && Number.isInteger(value) && value >= 0) {
    return value;
  }
  if (typeof value === "string" && value.trim()) {
    const n = Number(value);
    if (Number.isInteger(n) && n >= 0) return n;
  }
  return undefined;
}

/** σÑæτ║ªσ¡ùµ«╡Σ╕║ image∩╝¢σà╝σ«╣ images µò░τ╗äσÅûΘªûσ╝á */
function resolveOptionImage(
  record: Record<string, unknown>,
): string | undefined {
  const single = readString(record.image);
  if (single) return single.trim();
  return toStringArray(record.images)[0];
}

function parseOptions(value: unknown): InteractionCardOption[] {
  if (!Array.isArray(value)) return [];
  return value.reduce<InteractionCardOption[]>((acc, item) => {
    if (item && typeof item === "object" && !Array.isArray(item)) {
      const record = item as Record<string, unknown>;
      const label = typeof record.label === "string" ? record.label.trim() : "";
      if (label) {
        const image = resolveOptionImage(record);
        acc.push({ label, ...(image ? { image } : {}) });
      }
    } else if (typeof item === "string" && item.trim()) {
      acc.push({ label: item.trim() });
    }
    return acc;
  }, []);
}

function parseInfo(value: unknown): InteractionCardInfo {
  if (!value || typeof value !== "object" || Array.isArray(value)) {
    return { scenarioImages: [], options: [] };
  }
  const record = value as Record<string, unknown>;
  return {
    cardType: readString(record.card_type),
    guidance: readString(record.guidance),
    scenarioTitle: readString(record.scenario_title),
    scenarioDescription: readString(record.scenario_description),
    scenarioImages: toStringArray(record.scenario_images),
    warning: readString(record.warning),
    optionsDescription: readString(record.options_description),
    options: parseOptions(record.options),
  };
}

 * Φºúσ╝ÇΣ┐íσ░üΣ╕èτÜä extension σ¡ùµ«╡∩╝ÜσÅ»Φâ╜µÿ» JSON σ¡ùτ¼ªΣ╕▓µêûσ╖▓Φºúµ₧Éσ»╣Φ▒íπÇé

function unwrapExtension(value: unknown): Record<string, unknown> | null {
  let obj: unknown = value;
  if (typeof value === "string") {
    if (!value.trim()) return null;
    obj = tryParse(value.trim());
  }
  if (!obj || typeof obj !== "object" || Array.isArray(obj)) {
    return null;
  }
  const record = obj as Record<string, unknown>;
  return isCardHost(record) ? record : null;
}

export function parseInteractionCardPayload(
  raw?: string | null,
): InteractionCardPayload | null {
  if (raw == null || typeof raw !== "string" || !raw.trim()) {
    return null;
  }
  if (raw === "[DONE]") {
    return null;
  }
  const parsed = tryParse(raw.trim());
  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
    return null;
  }
  const outer = parsed as Record<string, unknown>;
  // µë┐Φ╜╜σìíτëçµò░µì«τÜäσ»╣Φ▒í∩╝ÜΣ╝ÿσàêΣ┐íσ░üσåà extension∩╝îσà╢µ¼íσñûσ▒éµ£¼Φ║½
  const cardHost = unwrapExtension(outer.extension) ?? outer;

  const rawInfo =
    cardHost.interaction_card_info !== undefined
      ? cardHost.interaction_card_info
      : cardHost;
  const info = parseInfo(rawInfo);

  // clientId / card_type / ΘÇëΣ╕¡µÇüΣ╝ÿσàêσÅûσñûσ▒éΣ┐íσ░ü∩╝îσ¢₧ΘÇÇσìíτëçµë┐Φ╜╜σ»╣Φ▒í
  return {
    clientId: readString(outer.clientId) ?? readString(cardHost.clientId),
    cardType:
      info.cardType ??
      readString(outer.card_type) ??
      readString(cardHost.card_type),
    info,
    selectedLabel:
      readString(outer.selectedLabel) ?? readString(cardHost.selectedLabel),
    selectedIndex:
      readIndex(outer.selectedIndex) ?? readIndex(cardHost.selectedIndex),
  };
}

/** µÿ»σÉªΣ╕║σÅ»µ╕▓µƒôτÜäΣ║ñΣ║Æσìíτëç∩╝êΦç│σ░æΦªüµ£ëµáçΘóÿ / µÅÅΦ┐░ / ΘÇëΘí╣∩╝ë */
export function isRenderableInteractionCardPayload(
  payload: InteractionCardPayload | null,
): payload is InteractionCardPayload {
  if (!payload) return false;
  const { info } = payload;
  return (
    !!info.scenarioTitle ||
    !!info.scenarioDescription ||
    !!info.guidance ||
    info.options.length > 0
  );
}

export function isInteractionCardEnvelopeData(data: string): boolean {
  if (!data || data === "[DONE]") return false;
  const parsed = tryParse(data.trim());
  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
    return false;

  const record = parsed as Record<string, unknown>;
  if (record.type !== INTERACTION_CARD_SSE_EVENT) return false;
  return isRenderableInteractionCardPayload(parseInteractionCardPayload(data));
}
