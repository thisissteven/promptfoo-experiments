import { Message, MessageSender, MessageType } from "@/app/store/types";
import { tryParse } from "@/app/utils/object";
import React, { useCallback, useMemo } from "react";
import styles from "./Quoted.module.scss";
import SvgClose from "@/app/icons/close.svg";
import { Popover, Image } from "antd";
import { useTranslations } from "@/app/locales";
import { isForbiddenMsg } from "@/app/store/formatMessage";

export enum QuotedFrom {
  ChatInput = "chatInput",
  Msg = "msg",
}
export default function Quoted({
  quotedMsg,
  onClose,
  from,
  handleClick,
  isMobile,
  msgSender,
}: {
  quotedMsg: Message;
  onClose?: () => void;
  from: QuotedFrom;
  handleClick?: (quotedMsg: Message) => void;
  isMobile?: boolean;
  msgSender?: MessageSender;
}) {
  const t = useTranslations();
  // 1. σ«îσà¿τº╗ΘÖñτè╢µÇüσÆî useEffect∩╝îµö╣τö¿ useMemo Φ«íτ«ùσÇ╝
  const content = useMemo(
    () => tryParse(quotedMsg?.content),
    [quotedMsg?.content],
  );

  const { msgContent, quoteRedrawn } = useMemo(() => {
    if (!quotedMsg || !content) return { msgContent: "", quoteRedrawn: false };

    let contentText = "";
    let isRedrawn = false;

    if (content.fileName) {
      if (content.fileType === MessageType.Video) {
        contentText = `[${t.chat.video}]: ${content.fileName}`;
      } else if (content.fileType === MessageType.Image) {
        contentText = `[${t.chat.image}]: ${content.fileName}`;
      } else {
        contentText = `[${t.chat.file}]: ${content.fileName}`;
      }
    } else if (typeof content.text === "string") {
      if (quotedMsg.hasRedrawn) {
        isRedrawn = true;
        contentText = t.chat.quotedHaveRedrawMsg;
      } else {
        contentText = `${content.text}`;
      }
    }

    return { msgContent: contentText, quoteRedrawn: isRedrawn };
  }, [quotedMsg, content]);

  // 2. τí«Σ┐¥µëÇµ£ëΣ╛¥Φ╡ûΘí╣σ«îµò┤
  const quotedPreview = useMemo(() => {
    if (quotedMsg && isForbiddenMsg(quotedMsg)) {
      return (
        <div
          className={styles.quotedPreview}
          dangerouslySetInnerHTML={{ __html: t.riskMsg.content }}
        />
      );
    }
    if (from === QuotedFrom.ChatInput || !content) return null;

    if (content.fileType === MessageType.Video) {
      return (
        <video src={content.fileUrl} width={isMobile ? 200 : 400} controls />
      );
    }
    if (content.fileType === MessageType.Image) {
      return (
        <Image
          height={200}
          src={content.fileUrl}
          alt={content.fileName}
          preview={{ mask: t.chat.preview }}
        />
      );
    }
    if (content.fileName) {
      return <span className={styles.fileContainer}>{content.fileName}</span>;
    }
    if (!content.text) return null;

    return (
      <div
        className={styles.quotedPreview}
        dangerouslySetInnerHTML={{ __html: content.text }}
      />
    );
  }, [content, isMobile, from]); // µ╖╗σèá from Σ╛¥Φ╡û

  // 3. σ«ëσà¿σñäτÉå getPopupContainer
  const getPopupContainer = useCallback(() => {
    const container = document.querySelector('[class^="Chat_chatMessages__"]');
    return container ? (container as HTMLElement) : document.body;
  }, []);

  // 4. Σ╜┐τö¿ useCallback Σ╝ÿσîûτé╣σç╗σñäτÉå
  const handleCloseClick = useCallback(
    (e: React.MouseEvent) => {
      e.stopPropagation();
      onClose?.();
    },
    [onClose],
  );

  const handleContainerClick = useCallback(() => {
    handleClick?.(quotedMsg);
  }, [handleClick, quotedMsg]);

  if (!quotedMsg) return null;

  return (
    <div
      className={`${styles.chatInputQuoted} quoteMsgContainer ${styles[from]} ${styles[isMobile ? "mobile" : "pc"]}`}
    >
      <Popover
        trigger="hover"
        content={quotedPreview}
        getPopupContainer={getPopupContainer}
        placement={msgSender === MessageSender.USER ? "left" : "right"}
        // µ╖╗σèá key σ╝║σê╢Θçìµû░σê¢σ╗║ Popover σ╜ôσåàσ«╣σÅÿσîû
        key={`popover-${quotedMsg.id}`}
      >
        <div className={styles.container} onClick={handleContainerClick}>
          {typeof onClose === "function" && (
            <SvgClose className={styles.closeIcon} onClick={handleCloseClick} />
          )}
          <span className={styles.divider}></span>
          {!quoteRedrawn && (
            <span className={styles.reply}>{t.chat.reply}: </span>
          )}
          {isForbiddenMsg(quotedMsg) ? (
            <span className={styles.mainContent}>{t.riskMsg.content}</span>
          ) : (
            <span className={styles.mainContent}>{msgContent}</span>
          )}
        </div>
      </Popover>
    </div>
  );
}

interface IProps {
  msg: Message;
}
export default function LineUp(props: IProps) {
  const t = useTranslations();
  const content = props.msg?.content as string;
  const chatStatus = useMessageStore((state) => state.chatStatus);

  useEffect(() => {
    props.msg.handleScrollToBottom?.(
      ScrollAction.FORCE_TO_BOTTOM,
      ScrollFrom.LINE_UP,
    );
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  if (!content || chatStatus !== CurrentChatStatus.LineUp) {
    return;
  }

  const getLineInfo = () => {
    let { text } = tryParse(content) || {};
    text = text.replace(/\d+/g, (num: number) => {
      return `<span style="color:#00ae42; font-weight: bold; margin: 0 4px">${num}</span>`;
    });
    return text;
  };

  const handleCancelLineUp = async () => {
    reportScene(ReportScene.CANCEL_LINE_UP);
    await cancelQueue();
    webSocketService.sendMessage(
      JSON.stringify({ text: t.chat.canceledLineUp }),
      {
        noNeedSendToWebsocket: true,
        senderType: MessageSender.USER,
        msgType: MessageType.Template,
      },
    );
  };

  return (
    <div
      className={`${styles.lineUp} ${isMobileDevice() ? styles.mobile : ""}`}
    >
      <p
        className={styles.title}
        dangerouslySetInnerHTML={{ __html: getLineInfo() }}
      ></p>
      <AppButton
        type="primary"
        onClick={handleCancelLineUp}
        theme="borderless"
        className={styles.cancel}
        extraclassname="extra"
      >
        {t.chat.cancelLineUp}
      </AppButton>
    </div>
  );

import { Message } from "@/app/store/types";
import React from "react";
import styles from "./File.module.scss";
import { parseFileData } from "@/app/utils/file";
import SvgFile from "@/app/icons/SvgFile.svg";
import { getHandyAppVersion, isHandyApp } from "@/app/utils/device";
import { copyToClipboard } from "@/app/utils/string";
import { useTranslations } from "@/app/locales";
import { message } from "antd";
import { openBrowser } from "@/app/utils/handyBridge";
import { isValidUrl } from "@/app/utils/url";

export default function File({ msg }: { msg: Message }) {
  const t = useTranslations();
  const content = msg.content as string;
  const openFile = async () => {
    const fileUrl = parseFileData(content)?.fileUrl;
    if (!isValidUrl(fileUrl)) {
      return;
    }
    if (isHandyApp()) {
      const handyVersion = getHandyAppVersion();
      if (
        handyVersion &&
        !!Number(handyVersion) &&
        Number(handyVersion) >= 5140 &&
        window.wx &&
        typeof window.wx.postMessage === "function"
      ) {
        openBrowser(fileUrl);
        return;
      }
      await copyToClipboard(fileUrl);
      message.success(t.tips.copyFileSuccess);
      return;
    }
    window.open(parseFileData(content)?.fileUrl);
  };
  return (
    <div
      onClick={openFile}
      className={`${styles.fileBox} ${msg.id ? styles.success : styles.error}`}
    >
      <span>{parseFileData(content)?.fileName}</span> <SvgFile />
    </div>
  );
}

interface MallContextCardProps {
  payload?: unknown;
  msg?: Message;
  timestamp?: number;
}

export default function MallContextCard({
  payload,
  msg,
  timestamp,
}: MallContextCardProps) {
  const t = useTranslations();
  const source = payload ?? msg?.content;
  const card = useMemo(() => parseMallContextCard(source), [source]);
  const imageUrl =
    card?.cardType === "product"
      ? card.productImage
      : card?.products[0]?.productImage;
  const [showImage, setShowImage] = useState(!!imageUrl);
  const [jumping, setJumping] = useState(false);
  const jumpingRef = useRef(false);

  useEffect(() => {
    setShowImage(!!imageUrl);
  }, [imageUrl]);

  if (!card) return null;

  const viewModel = buildViewModel(card, t.mallContextCard);
  const timeText = formatTimestamp(msg?.timestamp ?? timestamp ?? 0);

  const handleJump = async () => {
    if (jumpingRef.current) return;
    const context = getMallCardJumpContext(card);
    if (!context) {
      message.error(t.mallContextCard.jumpFailed);
      return;
    }
    console.log("context------>", context);
    jumpingRef.current = true;
    setJumping(true);
    try {
      const cachedUrl = msg?.jumpUrl;
      const jumpUrl =
        cachedUrl && isValidUrl(cachedUrl)
          ? cachedUrl
          : await fetchMallContextJumpUrl(context);
      if (!cachedUrl) {
        useMessageStore
          .getState()
          .patchMessage({ id: msg?.id, clientId: msg?.clientId }, { jumpUrl });
      }
      openMallJumpUrl(jumpUrl);
    } catch (error) {
      console.error("[MallCard] jump url failed:", error);
      message.error(t.mallContextCard.jumpFailed);
    } finally {
      jumpingRef.current = false;
      setJumping(false);
    }
  };

  return (
    <div className={styles.wrap}>
      {timeText && <div className={styles.time}>{timeText}</div>}
      <div
        className={styles.card}
        data-card-type={card.cardType}
        role="button"
        tabIndex={0}
        aria-busy={jumping}
        onClick={handleJump}
        onKeyDown={(event) => {
          if (event.key === "Enter" || event.key === " ") {
            event.preventDefault();
            void handleJump();
          }
        }}
      >
        <div className={styles.thumbnail}>
          {showImage && imageUrl ? (
            // eslint-disable-next-line @next/next/no-img-element
            <img
              src={imageUrl}
              alt={viewModel.productName}
              onError={() => setShowImage(false)}
            />
          ) : (
            <span>{viewModel.fallbackText}</span>
          )}
        </div>
        <div className={styles.content}>
          <div className={styles.title}>{viewModel.title}</div>
          <div className={styles.subtitle}>
            {viewModel.subtitleParts.map((part, index) => (
              <span key={`${part}-${index}`}>
                {index > 0 && <i aria-hidden="true">|</i>}
                {part}
              </span>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
}

const openMallJumpUrl = (jumpUrl: string) => {
  window.open(jumpUrl, "_blank", "noopener,noreferrer");
};

type MallContextCardTranslations = ReturnType<
  typeof useTranslations
>["mallContextCard"];

const buildViewModel = (
  card: MallContextCardData,
  translations: MallContextCardTranslations,
) => {
  if (card.cardType === "product") {
    return {
      productName: card.productName,
      fallbackText: getProductFallbackText(card),
      title: `${translations.consultingPrefix}${card.productName}`,
      subtitleParts: [
        translations.productConsultation,
        ...(card.skuName ? [card.skuName] : []),
      ],
    };
  }

  const product = card.products[0];
  const status =
    card.orderStatusText ||
    translations.orderStatus[
      card.orderStatus as keyof typeof translations.orderStatus
    ];
  const productSummary = product
    ? `${product.productName}${product.quantity ? ` x${product.quantity}` : ""}`
    : "";

  return {
    productName: product?.productName || translations.orderConsultation,
    fallbackText: translations.orderPlaceholder,
    title: `${translations.consultingPrefix}${translations.orderPrefix}${getMaskedOrderNo(card)}`,
    subtitleParts: [
      translations.orderConsultation,
      ...(status ? [status] : []),
      ...(productSummary ? [productSummary] : []),
    ],
  };
};

import type { MouseEvent } from "react";
import { RobotMessageExtensionType } from "@/app/store/types";
import { tryParse } from "@/app/utils/object";

export const GENERAL_INQUIRY_PATH_SUFFIX = "/my/support/general-inquiry";
const CREATE_ORDER_TAG = "createOrder";

/** delta µûçµ£¼Σ╕¡τÜä general-inquiry https Θô╛µÄÑ∩╝êσÉ½ QA σƒƒσÉì∩╝ë */
const GENERAL_INQUIRY_HTTPS_URL_RE =
  /https:\/\/[^\s<>"')\]]+\/my\/support\/general-inquiry[^\s<>"')\]]*/gi;

export function isHandoffContextExtension(extension?: string): boolean {
  if (!extension?.trim()) {
    return false;
  }
  const parsed = tryParse(extension.trim()) as { type?: string } | null;
  return parsed?.type === RobotMessageExtensionType.TransferCreateOrderLink;
}

export function resolveHandoffAiConversationId(
  extension?: string,
  aiConversationId?: string,
): string | undefined {
  const topLevel = aiConversationId?.trim();
  if (topLevel) {
    return topLevel;
  }
  if (!extension?.trim()) {
    return undefined;
  }
  const parsed = tryParse(extension.trim()) as {
    aiConversationId?: string;
  } | null;
  return parsed?.aiConversationId?.trim() || undefined;
}

export function isGeneralInquiryHttpsUrl(url: string): boolean {
  try {
    const linkUrl = new URL(url);
    return (
      linkUrl.protocol === "https:" &&
      linkUrl.pathname.includes(GENERAL_INQUIRY_PATH_SUFFIX)
    );
  } catch {
    return false;
  }
}

/** Σ╕║ general-inquiry https Θô╛µÄÑΦ┐╜σèá id=aiConversationId∩╝êσÅè handoff σ£║µÖ»τÜä tag∩╝ë */
export function appendHandoffParamsToGeneralInquiryUrl(
  rawUrl: string,
  conversationId: string,
  options?: { withCreateOrderTag?: boolean },
): string {
  if (!conversationId.trim() || !isGeneralInquiryHttpsUrl(rawUrl)) {
    return rawUrl;
  }
  try {
    const linkUrl = new URL(rawUrl);
    if (!linkUrl.searchParams.has("id")) {
      linkUrl.searchParams.set("id", conversationId.trim());
    }
    if (options?.withCreateOrderTag) {
      linkUrl.searchParams.set("tag", CREATE_ORDER_TAG);
    }
    return linkUrl.toString();
  } catch {
    return rawUrl;
  }
}

/** σ░å delta µûçµ£¼Σ╕¡τÜä general-inquiry Φú╕Θô╛µÄÑµö╣σåÖΣ╕║σ╕ª id τÜäΘô╛µÄÑ */
export function appendHandoffIdToGeneralInquiryUrlsInContent(
  content: string,
  conversationId?: string,
  options?: { withCreateOrderTag?: boolean },
): string {
  if (!content || !conversationId?.trim()) {
    return content;
  }
  return content.replace(GENERAL_INQUIRY_HTTPS_URL_RE, (matched) =>
    appendHandoffParamsToGeneralInquiryUrl(matched, conversationId, options),
  );
}

/** general-inquiry https Θô╛µÄÑτé╣σç╗∩╝ÜΦ┐╜σèá id=aiConversationId */
export function handleHandoffGeneralInquiryLinkClick(
  event: MouseEvent<HTMLElement>,
  options: {
    extension?: string;
    aiConversationId?: string;
  },
): void {
  const conversationId = resolveHandoffAiConversationId(
    options.extension,
    options.aiConversationId,
  );
  if (!conversationId) {
    return;
  }

  const anchor = (event.target as HTMLElement).closest("a");
  if (!anchor) {
    return;
  }

  const rawHref = anchor.getAttribute("href");
  if (!rawHref) {
    return;
  }

  if (!isGeneralInquiryHttpsUrl(rawHref)) {
    return;
  }

  const hrefWithParams = appendHandoffParamsToGeneralInquiryUrl(
    rawHref,
    conversationId,
    { withCreateOrderTag: isHandoffContextExtension(options.extension) },
  );
  if (hrefWithParams === rawHref) {
    return;
  }

  event.preventDefault();
  event.stopPropagation();

  if (anchor.target === "_blank") {
    window.open(hrefWithParams, "_blank", "noopener,noreferrer");
  } else {
    window.location.assign(hrefWithParams);
  }
}

function normalizeFormTypeKey(formType: unknown): string {
  if (typeof formType !== "string" || !formType.trim()) return "";
  return formType.trim().toLowerCase().replace(/_/g, "");
}

export function isSellingLogisticsFormType(formType: unknown): boolean {
  return normalizeFormTypeKey(formType) === "sellinglogistics";
}

export function isSellingOrdersFormType(formType: unknown): boolean {
  return normalizeFormTypeKey(formType) === "sellingorders";
}

function resolveSellingFormExtensionRecord(
  extension: string,
): Record<string, unknown> | null {
  const parsed = tryParse(extension.trim());
  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
    return null;
  }
  const outer = parsed as Record<string, unknown>;

  const nestedExtension = outer.extension;
  if (typeof nestedExtension === "string" && nestedExtension.trim()) {
    const inner = tryParse(nestedExtension.trim());
    if (inner && typeof inner === "object" && !Array.isArray(inner)) {
      return { ...(inner as Record<string, unknown>), ...outer };
    }
  }

  return outer;
}

function isSubmittedFlag(value: unknown): boolean {
  return value === true || value === "true" || value === 1;
}

function parseFieldSource(value: unknown): SellingFormFieldSource | undefined {
  return value === "selected" || value === "manual" ? value : undefined;
}

function parseLogisticsSubmittedPayload(
  payload: unknown,
): SellingLogisticsSubmittedPayload | undefined {
  if (!payload || typeof payload !== "object" || Array.isArray(payload)) {
    return undefined;
  }
  const p = payload as Record<string, unknown>;
  const orderId = typeof p.orderId === "string" ? p.orderId.trim() : undefined;
  const orderChannel =
    typeof p.orderChannel === "string" ? p.orderChannel.trim() : undefined;
  const trackingNumber =
    typeof p.trackingNumber === "string" ? p.trackingNumber.trim() : undefined;
  const orderSource = parseFieldSource(p.orderSource);
  const trackingSource = parseFieldSource(p.trackingSource);
  if (!orderId && !orderChannel && !trackingNumber) {
    return undefined;
  }
  return {
    orderId,
    orderChannel,
    trackingNumber,
    orderSource,
    trackingSource,
  };
}

function parseOrdersSubmittedPayload(
  payload: unknown,
): SellingOrdersSubmittedPayload | undefined {
  if (!payload || typeof payload !== "object" || Array.isArray(payload)) {
    return undefined;
  }
  const p = payload as Record<string, unknown>;
  const orderId = typeof p.orderId === "string" ? p.orderId.trim() : undefined;
  const orderChannel =
    typeof p.orderChannel === "string" ? p.orderChannel.trim() : undefined;
  const product = typeof p.product === "string" ? p.product.trim() : undefined;
  const products = parseSubmittedProducts(p.products);
  const orderSource = parseFieldSource(p.orderSource);
  const productSource = parseFieldSource(p.productSource);
  if (!orderId && !orderChannel && !product && products.length === 0) {
    return undefined;
  }
  return {
    orderId,
    orderChannel,
    product,
    ...(products.length > 0 ? { products } : {}),
    orderSource,
    productSource,
  };
}

function readFormMetaFromExtension(
  extension: string,
): Partial<SellingFormMeta> {
  const raw = resolveSellingFormExtensionRecord(extension);
  if (!raw) {
    return {};
  }
  const aiConversationId =
    typeof raw.aiConversationId === "string" ? raw.aiConversationId.trim() : "";
  const formMessageClientId =
    typeof raw.clientId === "string" ? raw.clientId.trim() : "";
  return {
    ...(aiConversationId ? { aiConversationId } : {}),
    ...(formMessageClientId ? { formMessageClientId } : {}),
  };
}

export function parseSellingFormMeta(
  source?: string | null | SellingFormMetaSource,
): SellingFormMeta | null {
  const input: SellingFormMetaSource =
    source == null || typeof source === "string"
      ? { extension: source }
      : source;

  let aiConversationId = "";
  let formMessageClientId = "";

  if (input.extension?.trim()) {
    const fromExt = readFormMetaFromExtension(input.extension.trim());
    aiConversationId = fromExt.aiConversationId ?? "";
    formMessageClientId = fromExt.formMessageClientId ?? "";
  }

  if (!aiConversationId && typeof input.aiConversationId === "string") {
    aiConversationId = input.aiConversationId.trim();
  }
  if (!formMessageClientId && typeof input.formMessageClientId === "string") {
    formMessageClientId = input.formMessageClientId.trim();
  }

  if (!aiConversationId || !formMessageClientId) {
    return null;
  }
  return { aiConversationId, formMessageClientId };
}

export function parseSellingLogisticsExtension(
  extension?: string | null,
): SellingLogisticsExtension | null {
  if (extension == null || typeof extension !== "string" || !extension.trim()) {
    return null;
  }

  const raw = resolveSellingFormExtensionRecord(extension);
  if (!raw || !isSellingLogisticsFormType(raw.formType)) {
    return null;
  }

  const submittedPayload = parseLogisticsSubmittedPayload(
    raw.submittedPayload ?? raw.payload,
  );
  const submitted =
    isSubmittedFlag(raw.submitted) || submittedPayload !== undefined;
  return {
    formType: String(raw.formType).trim(),
    submitted,
    submissionId:
      typeof raw.submissionId === "string" ? raw.submissionId : undefined,
    submittedPayload,
  };
}

export function parseSellingOrdersExtension(
  extension?: string | null,
): SellingOrdersExtension | null {
  if (extension == null || typeof extension !== "string" || !extension.trim()) {
    return null;
  }

  const raw = resolveSellingFormExtensionRecord(extension);
  if (!raw || !isSellingOrdersFormType(raw.formType)) {
    return null;
  }

  const submittedPayload = parseOrdersSubmittedPayload(
    raw.submittedPayload ?? raw.payload,
  );
  const submitted =
    isSubmittedFlag(raw.submitted) || submittedPayload !== undefined;
  return {
    formType: String(raw.formType).trim(),
    submitted,
    submissionId:
      typeof raw.submissionId === "string" ? raw.submissionId : undefined,
    submittedPayload,
  };
}

export function parseSellingFormExtension(
  extension?: string | null,
):
  | ({ variant: "logistics" } & SellingLogisticsExtension)
  | ({ variant: "orders" } & SellingOrdersExtension)
  | null {
  const logistics = parseSellingLogisticsExtension(extension);
  if (logistics) {
    return { variant: "logistics", ...logistics };
  }
  const orders = parseSellingOrdersExtension(extension);
  if (orders) {
    return { variant: "orders", ...orders };
  }
  return null;
}

function transformATag(html: string) {
  return html.replace(
    /<a([\s\S]*?)href="([\s\S]*?)"([\s\S]*?)>/gi,
    // /<a([^>]*?)href="([^"]*)"([^>]*?)>/gi,

      const newHref =
        isApp() && !shouldUseWebview(href)
          ? `bambulab://bbl/webview?urlString=${encodeURIComponent(href)}`
          : href;
      const hasTarget = /\starget\s*=/i.test(before + after);
      const targetStr = hasTarget ? "" : ' target="_blank"';
      return `<a${before}href="${newHref}"${after}${targetStr}>`;
    },
  );
}

const getClientId = (query: MallCardQuery) => {
  const storageKey = `mall_card_snapshot_client_id:${JSON.stringify(query)}`;
  const stored = sessionStorage.getItem(storageKey);
  if (stored) return { clientId: stored, storageKey };

  const clientId = `mall-${query.cardType}-${createMessageClientId()}`;
  sessionStorage.setItem(storageKey, clientId);
  return { clientId, storageKey };
};

const hasRenderedCard = (clientId: string) => {
  const { historyMsgList, msgList } = useMessageStore.getState();
  return [...historyMsgList, ...msgList].some(
    (item) => item.clientId === clientId,
  );
};

/**
 * σòåσƒÄσ╕ªσÅéΦ┐¢τ║┐σç║σìí∩╝ÜΦºúµ₧É URL Σ╕èΣ╕ïµûç ΓåÆ Φ░â snapshot ΓåÆ µÅÆσàÑσìíτëçµ░öµ│í ΓåÆ µ╕àµÄëµ£¼µ¼íτö¿Φ┐çτÜä queryπÇé
 * Φ»╖µ▒éσñ▒Φ┤Ñµù╢Σ┐¥τòÖ query Σ╕Ä clientId∩╝îΣ╛┐Σ║Äτö¿µê╖σê╖µû░ΘçìΦ»òπÇé
 */
export const renderMallCardFromUrl = async (
  searchParams?: URLSearchParams,
): Promise<void> => {
  if (typeof window === "undefined") return;
  // µ£¬τÖ╗σ╜òΣ╕ìΦ░âµÄÑσÅúπÇüΣ╣ƒΣ╕ìµ╕àσÅé∩╝êτÖ╗σ╜òσ¢₧Φ╖│σÉÄΣ╗ìΘ£Çσ╕ªτ¥ÇΣ╕èΣ╕ïµûçσç║σìí∩╝ë
  if (!authService.getToken()) return;

  const params =
    searchParams ?? new URLSearchParams(window.location.search || "");
  const query = readQuery(params);
  if (!query) return;
  const { clientId, storageKey } = getClientId(query);
  if (hasRenderedCard(clientId)) {
    clearMallContextQuery();
    sessionStorage.removeItem(storageKey);
    return;
  }

  try {
    const card = await createMallCardSnapshot({ ...query, clientId });
    const responseClientId = card.clientId || clientId;
    if (!hasRenderedCard(responseClientId)) {
      useMessageStore.getState().setRobotMessage({
        clientId: responseClientId,
        type:
          card.cardType === "product"
            ? MessageType.MallProductCard
            : MessageType.MallOrderCard,
        headerType: HeaderType.MESSAGE,
        senderType: MessageSender.ROBOT,
        messageType: RobotMessageType.ROBOT_NORMAL_REPLY,
        content: card.cardType === "product" ? "σòåσôüσìíτëç" : "Φ«óσìòσìíτëç",
        extension: JSON.stringify(card),
        timestamp: Date.now(),
        senderName: "µÖ║Φâ╜σè⌐µëï",
        robotData: {},
      });
    }
    clearMallContextQuery();
    sessionStorage.removeItem(storageKey);
  } catch (error) {
    console.error("[MallCard] snapshot failed:", error);
  }
};
