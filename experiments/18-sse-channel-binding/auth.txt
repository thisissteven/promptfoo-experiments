interface AuthContextType {
  isAuthenticated: boolean;

export const deleteUnUsedParams = (url: URL) => {
  try {
    // Θâ¿σêåσàÑσÅúΣ╝Üµï╝σç║ `??timestamp=...`∩╝îsearchParams τÜä key Σ╝ÜσÅÿµêÉ `?timestamp`∩╝î
    // σÅ¬ delete("timestamp") σêáΣ╕ìµÄë∩╝îσ┐àΘí╗σàêµèèσñÜΣ╜ÖτÜä `?` σ╜ÆΣ╕ÇµÄëπÇé
    if (url.search.startsWith("??")) {
      url.search = url.search.slice(1);
    }
    url.searchParams.delete("token");
    url.searchParams.delete("timestamp");
    url.searchParams.delete("?token");
    url.searchParams.delete("?timestamp");
    window.history.replaceState({}, "", url.toString());
  } catch (error) {
    console.error("Failed to delete unused params:", error);
  }
};

const deleteUrlParams = (url: URL, ...paramNames: string[]) => {
  try {
    paramNames.forEach((name) => url.searchParams.delete(name));
    window.history.replaceState({}, "", url.toString());
  } catch (error) {
    console.error("Failed to delete url params:", error);
  }
};

export function AuthProvider({ children }: { children: ReactNode }) {
  const [isAuthenticated, setIsAuthenticated] = useState(false);
  const [user, setUser] = useState<User | null>(null);
  const [isLoading, setIsLoading] = useState(true);

      const url = new URL(window.location.href);
      const ticket = url.searchParams.get("ticket");
      const rawBblToken = url.searchParams.get("bblToken")?.trim();
      deleteUnUsedParams(url);
      const bblToken = await resolveBblTokenFromQuery(rawBblToken);
      if (bblToken) {
        authService.setBblToken(bblToken);
      }
      if (rawBblToken) {
        deleteUrlParams(url, "bblToken");
      }
      if (ticket) {

        const { login } = await import("../api/auth");
        const res = await login(ticket);
        // Σ┐¥σ¡ÿ Token σÆîτö¿µê╖Σ┐íµü»
        authService.setToken(res.accessToken, res);
        authService.setUserInfo(res);
        setUser(res.cusInfo);
        setIsAuthenticated(true);

        // µ╕àΘÖñ URL Σ╕¡τÜä ticket σÅéµò░
        deleteUrlParams(url, "ticket", "redirect", "bblToken");
      } else {
        // 2. µúÇµƒÑµÿ»σÉªµ£ëσ╖▓σ¡ÿσ£¿τÜä Token
        const existingToken = authService.getToken();
        console.log("existingToken------>", existingToken);
        if (existingToken) {
          // Θçìµû░Φ░âτö¿setTokenτí«Σ┐¥τÖ╗σ╜òtokenΦó½Φ«╛τ╜«
          authService.setToken(existingToken);
          // µ£ë Token∩╝îσ░¥Φ»òτ╗¡µ£ƒ∩╝êσªéµ₧£Θ£ÇΦªü∩╝ë
          await authService.renewalToken();
          // Σ╗Ä LocalStorage ΦÄ╖σÅûτö¿µê╖Σ┐íµü»
          const loginInfo = localStorage.getItem("login_info");
          if (loginInfo) {
            try {
              const info = JSON.parse(loginInfo);
              setUser(info.cusInfo);
              authService.setUserInfo(info);
              setIsAuthenticated(true);
            } catch (error) {
              console.error("Failed to parse login info:", error);
            }
          } else {
            setIsAuthenticated(true);
          }
        } else {
          // µùá IM Token µù╢∩╝îURL σ╕ª bblToken σêÖτö¿σà╢µìóσÅûτÖ╗σ╜òµÇü∩╝êσòåσƒÄτ¡ëσñûΘâ¿σàÑσÅú∩╝ë

          // σªéµ₧£µÿ»σ£¿handy appΣ╕¡∩╝îΦ╡░µùáµäƒτƒÑτÖ╗σ╜ò
          if (isHandyApp()) {
            const res = await authService.fetchHandyAppLogin();
            if (res) {
              setUser(res.cusInfo);
              authService.setUserInfo(res);
              setIsAuthenticated(true);
            } else {
              setIsAuthenticated(false);
              setUser(null);
            }
            return;
          } else if (bblToken) {
            const res = await authService.loginWithBBLToken(bblToken);
            authService.setUserInfo(res);
            setUser(res.cusInfo);
            setIsAuthenticated(true);
            return;
          }
          // σòåσƒÄτ¡ëµ¥Ñµ║É∩╝êfrom=6∩╝ëΦªüµ▒éσ┐àΘí╗τÖ╗σ╜ò∩╝Üµùá Token µù╢τ¢┤µÄÑΦ╖│Φ╜¼σ«ÿτ╜æτÖ╗σ╜ò∩╝î
          // ΦÇîΘ¥₧Φ┐¢σàÑµ╕╕σ«óµ¿íσ╝ÅπÇélogin() Σ╝ÜΘçìσ«ÜσÉæσ╜ôσëìΘí╡Θ¥óπÇé
          if (isMustLoginFromSource()) {
            await authService.login();
            return;
          }
          // µùá Token∩╝îµ╕╕σ«óµ¿íσ╝Å
          setIsAuthenticated(false);
          setUser(null);
          authService.setUserInfo(null);
        }
      }
    } catch (error) {
      console.error("Auth check failed:", error);
      setIsAuthenticated(false);
      setUser(null);
      authService.setUserInfo(null);
    } finally {
      setIsLoading(false);
    }
  }, []);

  /**
   * Σ╕╗σè¿τÖ╗σ╜ò
   * τö¿µê╖τé╣σç╗τÖ╗σ╜òµîëΘÆ«µù╢Φ░âτö¿
   */
  const login = useCallback(
    async (target: string, extraParams?: Record<string, string>) => {
      // Φ░âτö¿ AuthService τÜäτÖ╗σ╜òµû╣µ│ò
      // Σ╝ÜΦç¬σè¿Θçìσ«ÜσÉæσê░σ«ÿτ╜æτÖ╗σ╜òΘí╡
      await authService.login(false, target, extraParams);
    },
    [],
  );

  /**
   * τÖ╗σç║
   */
  const logout = useCallback(() => {
    authService.clearToken();
    setIsAuthenticated(false);
    setUser(null);

    // σê╖µû░Θí╡Θ¥ó
    window.location.reload();
  }, []);

  /**
   * µëïσè¿Φ«╛τ╜«Φ«ñΦ»üτè╢µÇü
   * τö¿Σ║Ä HandyApp σ£║µÖ»∩╝îσ£¿ΦÄ╖σÅûσê░τö¿µê╖Σ┐íµü»σÉÄµ¢┤µû░τÖ╗σ╜òµÇü
   */
  const setAuthenticated = useCallback((authenticated: boolean) => {
    setIsAuthenticated(authenticated);
  }, []);

  // σê¥σºïσîûµù╢µúÇµƒÑΦ«ñΦ»üτè╢µÇü
  useEffect(() => {
    checkAuth();
  }, [checkAuth]);

  // τÖ╗σ╜òµêÉσèƒσÉÄΘóäµïëσÅûσö«Σ╕¡τë⌐µ╡üΦ«óσìòσêùΦí¿
  useEffect(() => {
    if (isLoading) return;
    const { fetchLogisticsOrders, resetLogisticsOrders } =
      useSellingLogisticsOrdersStore.getState();
    if (isAuthenticated && isSellingLogisticsOrdersPrefetchEnabled()) {
      void fetchLogisticsOrders();
    } else {
      resetLogisticsOrders();
    }
  }, [isAuthenticated, isLoading]);

  // τ¢æσÉ¼ AuthService τÜäτÖ╗σç║Σ║ïΣ╗╢
  useEffect(() => {
    const handleLogout = () => {
      setIsAuthenticated(false);
      setUser(null);
    };

    const handleTokenExpired = () => {
      console.warn("Token expired, user logged out");
      setIsAuthenticated(false);
      setUser(null);
    };

    // Φ«óΘÿàΣ║ïΣ╗╢
    const unsubscribeLogout = authService.on(
      AuthEventType.LOGOUT,
      handleLogout,
    );
    const unsubscribeTokenExpired = authService.on(
      AuthEventType.TOKEN_EXPIRED,
      handleTokenExpired,
    );

    // µ╕àτÉåΦ«óΘÿà
    return () => {
      unsubscribeLogout();
      unsubscribeTokenExpired();
    };
  }, []);

  const value: AuthContextType = {
    isAuthenticated,
    user,
    isLoading,
    login,
    logout,
    checkAuth,
    setAuthenticated,
  };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}
