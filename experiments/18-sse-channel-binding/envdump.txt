function isNetworkError(error: unknown): boolean {
  return (
    error instanceof TypeError &&
    /load failed|failed to fetch|networkerror/i.test(error.message)
  );
}

function parseJsonPreservingBigInt(text: string, fields: string[]) {
  const pattern = new RegExp(`"(${fields.join("|")})"\\s*:\\s*(-?\\d+)`, "g");
  const safeText = text.replace(pattern, '"$1":"$2"');
  return JSON.parse(safeText);
}
