  const storedChannel = getStoredChannel();

    return storedChannel;

  }
  if (isHandyApp()) {
    return SOCKET_CONNECT_CHANNEL.HANDY_OR;

  return SOCKET_CONNECT_CHANNEL.OTHERS_OR;
}

const handleStoreBoundChannel = async (
  deps: StoreBoundChannelDeps,
): Promise<boolean> => {
  if (!hasStoreBoundParams()) return false;

  if (!deps.isAuthenticated) {
    authService.login(true);
    return true;
  }

  const storedChannel = getStoredChannel() || "";
  const storedStore = getStoredStore() || "";

  // σ╖▓µ£ëτ╝ôσ¡ÿΣ╝ÿσàêτö¿∩╝îΘü┐σàìµ»Åµ¼íΦ┐¢Θí╡Θ¥óΘâ╜µëôµÄÑσÅú∩╝¢τ╝ôσ¡ÿσÅ¬µ£ëΦ╡░Φ┐çµêÉσèƒΦ╖»σ╛äµëìΣ╝ÜσåÖσàÑ
  let issueTypeCache = getStoredIssueTypeCache();
  if (!issueTypeCache) {
    try {
      const reception = await getStoreReception({
        store: storedStore,
        channel: storedChannel,
      });
      if (!reception || reception.blocked || !reception.issueTypeId) {
        // Θ¥ÖΘ╗ÿσ¢₧ΘÇÇ∩╝Üblocked=true / µÄÑσÅúΦ┐öσ¢₧τ⌐║ issueTypeId Σ╕Çσ╛ïΦ╡░µÖ«ΘÇÜµ╡üτ¿ï∩╝î
        // Σ╕ìµ╢êΦ┤╣σÉÄτ½»Φ┐öσ¢₧τÜä fallbackIssueTypeId∩╝êΣ║ºσôüσå│τ¡û Q2=a∩╝ë
        console.warn(
          "[StoreBound] reception blocked or empty, fallback to default flow:",
          reception?.blockReason,
        );
        return false;
      }
      issueTypeCache = {
        issueTypeId: reception.issueTypeId,
        issueTypeName: reception.issueTypeName || "",
      };
      setStoredIssueTypeCache(issueTypeCache);
    } catch (error) {
      console.error("[StoreBound] getStoreReception failed:", error);
      return false;
    }
  }

  try {
    const sessionInfo = await getSessionStatus();
    deps.setSessionInfo(sessionInfo);

    await deps.loadHistory();

    const onSocketError = () => {
      console.error(
        `[StoreBound] socket initialization failed, channel=${storedChannel}, store=${storedStore}`,
      );
    };

    if (sessionInfo.hasActiveConversation) {
      // µ£ëΦ┐¢ΦíîΣ╕¡τÜäΣ╝ÜΦ»¥∩╝ÜΣ╗Ñµ£ìσèíτ½»Φ┐öσ¢₧τÜä issueTypeId Σ╕║σçå∩╝êτö¿µê╖Σ╕èµ¼íΣ╝ÜΦ»¥τö¿τÜä∩╝ë∩╝î
      // σÉîµ¡Ñσê╖µû░τ╝ôσ¡ÿ∩╝îΣ┐¥Φ»üσÉÄτ╗¡ΘçìσÅæ / IM Θô╛µÄÑσñìτö¿ issueTypeId Σ╕Äσ╜ôσëìΣ╝ÜΦ»¥Σ╕ÇΦç┤
      const activeIssueTypeId = sessionInfo.issueTypeId;
      if (activeIssueTypeId) {
        setStoredIssueTypeCache({
          issueTypeId: activeIssueTypeId,
          issueTypeName: issueTypeCache.issueTypeName,
        });
      }
      initSocket({ id: activeIssueTypeId }, (status) => {
        if (status === SocketConnectionStatus.ERROR) {
          onSocketError();
          return;
        }
        if (typeof deps.initSuccess === "function") {
          deps.initSuccess();
        }
      });
      if (sessionInfo.status === ConversationStatus.LineUp) {
        useMessageStore
          .getState()
          .setCurrentChatStatus(CurrentChatStatus.LineUp);
        webSocketService.sendMessage(
          JSON.stringify({ text: sessionInfo.queueDesc }),
          {
            msgType: MessageType.LineUp,
            noNeedSendToWebsocket: true,
            senderType: MessageSender.SYSTEM,
          },
        );
      } else {
        useMessageStore
          .getState()
          .setCurrentChatStatus(CurrentChatStatus.Online);
      }
    } else {
      const canProceed = await deps.checkCanService();
      if (!canProceed) {
        deps.setRobotMessage({
          type: MessageType.RichText,
          headerType: HeaderType.MESSAGE,
          senderType: MessageSender.ROBOT,
          messageType: RobotMessageType.ROBOT_NORMAL_REPLY,
          showFeedFooter: false,
          content: deps.getCanServiceContent(),
          timestamp: Date.now(),
          senderName: "µÖ║Φâ╜σè⌐µëï",
          robotData: {},
        });
        return true;
      }

      if (typeof window !== "undefined") {
        window.globalData = window.globalData || {};
        window.globalData.serveSocketPreparedCallback = () => {
          useMessageStore
            .getState()
            .setCurrentChatStatus(CurrentChatStatus.SocketConnected);
          window.globalData!.serveSocketPreparedCallback = undefined;
        };
      }
      initSocket({ id: issueTypeCache.issueTypeId }, (status) => {
        if (status === SocketConnectionStatus.ERROR) {
          onSocketError();
          return;
        }
        if (typeof deps.initSuccess === "function") {
          deps.initSuccess();
        }
      });
    }
  } catch (error) {
    console.error(
      `[StoreBound] initialization failed, channel=${storedChannel}, store=${storedStore}:`,
      error,
    );
  }
  return true;
};

const fetchOneShotIssueType = async (
  params: StoreBoundLinkParams,
): Promise<{ issueTypeId: string; issueTypeName: string } | null> => {
  try {
    const reception = await getStoreReception({
      store: params.store,
      channel: params.channel,
    });
    if (!reception || reception.blocked || !reception.issueTypeId) {
      console.warn(
        "[StoreBoundOneShot] reception blocked or empty, fallback:",
        reception?.blockReason,
      );
      return null;
    }
    const issueTypeName = reception.issueTypeName || "";
    setOneShotIssueTypeName(issueTypeName);
    return { issueTypeId: reception.issueTypeId, issueTypeName };
  } catch (error) {
    console.error("[StoreBoundOneShot] getStoreReception failed:", error);
    return null;
  }
};

      useMessageStore
        .getState()
        .setCurrentChatStatus(CurrentChatStatus.SocketConnected);
      window.globalData!.serveSocketPreparedCallback = undefined;
    };

  initSocket(
    { id: issue.issueTypeId },

const handleOneShotDirectAfterLogin = async (
  params: StoreBoundLinkParams,
  deps: OneShotDirectAfterLoginDeps,
): Promise<boolean> => {
  const issue = await fetchOneShotIssueType(params);
  if (!issue) {
    // blocked / µÄÑσÅúσ╝éσ╕╕∩╝Üσ¢₧ΘÇÇµÖ«ΘÇÜσê¥σºïσîûµ╡üτ¿ï∩╝êΣ║ºσôüσå│τ¡û fallback=Θù«Θóÿτ▒╗σ₧ïµ░öµ│í∩╝ë
    return false;
  }

  const override = { channel: params.channel, store: params.store };
  try {
    const sessionInfo = await getSessionStatus();
    deps.setSessionInfo(sessionInfo);
    await deps.loadHistory();

    const onSocketError = () => {
      console.error(
        `[StoreBoundOneShot] socket init failed, channel=${params.channel}, store=${params.store}`,
      );
    };

    if (sessionInfo.hasActiveConversation) {
      const activeIssueTypeId = sessionInfo.issueTypeId;
      initSocket(
        { id: activeIssueTypeId },
        (status) => {
          if (status === SocketConnectionStatus.ERROR) {
            onSocketError();
            return;
          }
          if (typeof deps.initSuccess === "function") {
            deps.initSuccess();
          }
        },
        override,
      );
      if (sessionInfo.status === ConversationStatus.LineUp) {
        useMessageStore
          .getState()
          .setCurrentChatStatus(CurrentChatStatus.LineUp);
        webSocketService.sendMessage(
          JSON.stringify({ text: sessionInfo.queueDesc }),
          {
            msgType: MessageType.LineUp,
            noNeedSendToWebsocket: true,
            senderType: MessageSender.SYSTEM,
          },
        );
      } else {
        useMessageStore
          .getState()
          .setCurrentChatStatus(CurrentChatStatus.Online);
      }
    } else {
      const canProceed = await deps.checkCanService();
      if (!canProceed) {
        deps.setRobotMessage({
          type: MessageType.RichText,
          headerType: HeaderType.MESSAGE,
          senderType: MessageSender.ROBOT,
          messageType: RobotMessageType.ROBOT_NORMAL_REPLY,
          showFeedFooter: false,
          content: deps.getCanServiceContent(),
          timestamp: Date.now(),
          senderName: "µÖ║Φâ╜σè⌐µëï",
          robotData: {},
        });
        return true;
      }

      if (typeof window !== "undefined") {
        window.globalData = window.globalData || {};
        window.globalData.serveSocketPreparedCallback = () => {
          useMessageStore
            .getState()
            .setCurrentChatStatus(CurrentChatStatus.SocketConnected);
          window.globalData!.serveSocketPreparedCallback = undefined;
        };
      }
      initSocket(
        { id: issue.issueTypeId },
        (status) => {
          if (status === SocketConnectionStatus.ERROR) {
            onSocketError();
            return;
          }
          if (typeof deps.initSuccess === "function") {
            deps.initSuccess();
          }
        },
        override,
      );
    }
  } catch (error) {
    console.error(
      `[StoreBoundOneShot] init failed, channel=${params.channel}, store=${params.store}:`,
      error,
    );
    // σê¥σºïσîûσ╝éσ╕╕∩╝êgetSessionStatus / checkCanService τ¡ë∩╝ë∩╝ÜΦ┐öσ¢₧ false σ¢₧ΘÇÇµÖ«ΘÇÜσê¥σºïσîûµ╡üτ¿ï∩╝î
    // σÉªσêÖΦ░âτö¿µû╣Σ╝ÜΦ»»σêñΣ╕║σ╖▓σñäτÉåΦÇî return∩╝îσ»╝Φç┤τö¿µê╖ΘÖ╖σàÑµùáΣ╝ÜΦ»¥πÇüµùá socket τÜäτè╢µÇüπÇé
    return false;
  }
  return true;
};

const insertFormMessage = (searchParams?: URLSearchParams) => {
  if (!shouldRestoreSellingLogisticsSession(searchParams)) {
    return;

  const restored = restoreSellingLogisticsSessionToChat({
    addOutgoingMessage,
    setRobotMessage,
  });

    return;
  }

  void persistSellingLogisticsSessionMessages(cache).catch((err) => {
    console.warn("[insertFormMessage] saveRobotInfo failed:", err);
  });
};

  private async handleErrorMsgInsert() {

    const renderedClientIds = new Set(

    const errorFile = errorFileLocalStore.get() as Record<
      string,

    if (errorFile) {
      for (const fileId in errorFile) {

          webSocketService.sendMessage(
            `{"fileName":"${fileInfo.file.name}", "fileUrl": "${URL.createObjectURL(fileInfo.file)}", "fileId": "${fileId}"}`,

              clientId: fileId,
              fileInfo,
            },
          );
        }
        errorFileLocalStore.remove(fileId);
      }
    }
    for (const [, msg] of this.messageQueue) {
      if (
        msg.content &&
        msg.clientId &&
        msg.senderType === MessageSender.USER
      ) {
        if (!renderedClientIds.has(msg.clientId)) {
          webSocketService.sendMessage(msg.content, {
            msgType: msg.type,
            noNeedSendToWebsocket: true,
            senderType: msg.senderType,
            status: MessageStatus.ERROR,
            clientId: msg.clientId,
          });
        }
        this.messageQueue.delete(msg.clientId);
      }
    }

  private setupEventHandlers(): void {
    if (!this.socket) return;

    this.socket.onopen = async () => {
      this.reconnectCount = 0;
      this.updateStatus(SocketConnectionStatus.CONNECTED);
      this.startHeartbeat();
      this.handleReconnectCallback();
    };

    this.socket.onmessage = (event) => {
      // τ¢æµ╡ïσ┐âΦ╖│
      if (event.data === HeartType.PING || event.data === HeartType.PONG) {
        if (event.data === HeartType.PING) {
          this.socket?.send(HeartType.PONG);
        } else {
          this.handlePong();
        }
        return;
      }
      try {
        const message = this.parseMessage(event.data);
        if (!message) {
          this.options.onError?.(new ErrorEvent("message-parse-failed"));
          return;
        }
        this.options.onCommonMessageHandler?.(message);
        if (message.headerType === HeaderType.ACK) {
          if (message.type === MessageType.MsgEnd) {
            // σ╜ôµ╢êµü»τ╗ôµ¥ƒµù╢∩╝îµùáΘ£Çσåìµ¢┤µû░ack
            this.options.onMessage?.(message);
          } else {
            // σñäτÉåACKτí«Φ«ñ
            this.handleAck(message);
          }
        } else {
          // µö╢σê░Θ¥₧ACKµ╢êµü»µù╢σÅæΘÇüACKτí«Φ«ñ
          if (message.id) {
            this.sendAck(message);
          }

          // Θ¥₧ACKµ╢êµü»Σ╕èµèÑ
          reportScene(
            ReportScene.AGENT_MSG_RECEIVE,
            JSON.stringify({
              id: message.id,
              type: message.type,
              conversationId: message.conversationId,
            }),
          );

          // Φ░âτö¿onMessageσ¢₧Φ░â
          this.options.onMessage?.(message);
        }
      } catch (error) {
        console.error("µ╢êµü»σñäτÉåσñ▒Φ┤Ñ:", error);
        this.options.onError?.(new ErrorEvent("message-parse-failed"));
      }
    };

    this.socket.onerror = (event) => {

    };

          SocketErrorCode.BlackList,

          SocketErrorCode.CancelLineUpByUser,
          SocketErrorCode.CloseChatByUser,
          SocketErrorCode.LineUpTimeout,
          SocketErrorCode.ChatTimeout,
          SocketErrorCode.CloseChatByAgent,

          SocketErrorCode.EmptyAgentService,

          SocketErrorCode.ServiceLimitStream,
          // Σ╕Üσèíσ╝éσ╕╕
          SocketErrorCode.BusinessError,
        ].includes(event.code)

  }
  // σü£µ¡óσ┐âΦ╖│µúÇµ╡ï
  stopHeartbeat() {
    if (this.heartbeatInterval) {
      clearInterval(this.heartbeatInterval);
      this.heartbeatInterval = null;
    }

    if (this.pingTimeout) {
      clearTimeout(this.pingTimeout);
      this.pingTimeout = null;
    }
  }
  // σ╝Çσºïσ┐âΦ╖│µúÇµ╡ï
  startHeartbeat() {
    this.missedPongCount = 0;
    // σ╝Çσºïσ┐âΦ╖│
    this.sendPing();

    if (this.heartbeatInterval) {
      clearInterval(this.heartbeatInterval);
      this.heartbeatInterval = null;
    }

    // Φ«╛τ╜«σ«Üµù╢σÖ¿σ«Üµ£ƒσÅæΘÇüσ┐âΦ╖│
    this.heartbeatInterval = setInterval(() => {
      this.sendPing();
    }, this.heartInterval);
  }

  // σñäτÉåpongσôìσ║ö
  handlePong() {
    this.lastPongTime = Date.now();
    this.missedPongCount = 0;
    // µö╢σê░pongµëìΦ»┤µÿÄΦ┐Öµ¥íΦ┐₧µÄÑτ£ƒτÜäσÅ»τö¿∩╝îµ¡ñµù╢µëìσàüΦ«╕Θçìτ╜«σ┐âΦ╖│ΘçìΦ┐₧τÜäµ¼íµò░Σ╕èΘÖÉ
    this.heartbeatReconnectCount = 0;

    if (this.pingTimeout) {
      clearTimeout(this.pingTimeout);
      this.pingTimeout = null;
    }
  }

  /**
   * σ┐âΦ╖│Φ┐₧τ╗¡Φ╢àµù╢τÜäσà£σ║òσñäτÉåπÇé
   *
   * σìèσ╝ÇΦ┐₧µÄÑΣ╕ï readyState Σ╗ìµÿ» OPENπÇüΣ╣ƒΣ╕ìΣ╝ÜΦºªσÅæ onclose∩╝îΦïÑΣ╕ìΣ╕╗σè¿σ║ƒσ╝âΦ┐Öµ¥íΦ┐₧µÄÑ∩╝î
   * σÉÄτ╗¡µ╢êµü»Θâ╜Σ╝ÜΦó½ send Φ┐¢Θ╗æµ┤₧Σ╕öµï┐Σ╕ìσê░ ACK∩╝îτö¿µê╖Σ╛ºΦí¿τÄ░Σ╕║πÇîµ╢êµü»σÅæΣ╕ìσç║σÄ╗Σ╕öµùáΣ╗╗Σ╜òµÅÉτñ║πÇìπÇé
   */
  private handleHeartbeatDead(): void {
    if (this.isManualClose) return;

    this.stopHeartbeat();

    const deadSocket = this.socket;
    reportScene(
      ReportScene.SOCKET_CONNECT,
      JSON.stringify({
        reason: "heartbeat-timeout",
        missedPongCount: this.missedPongCount,
        lastPingTime: this.lastPingTime,
        lastPongTime: this.lastPongTime,
        readyState: deadSocket?.readyState ?? null,
      }),
    );
    reporterInstance.flush();

    this.socket = null;
    if (deadSocket) {
      // σàêµæÿσ¢₧Φ░âσåìσà│Θù¡∩╝Üσìèσ╝ÇΦ┐₧µÄÑτ¡ëΣ╕ìσê░µ£ìσèíτ½»τÜä close σôìσ║ö∩╝î
      // µ╡ÅΦºêσÖ¿µ£Çτ╗êΦíÑσÅæτÜä onclose Σ╝ÜσåìΦ╡░Σ╕Çµ¼í handleReconnect∩╝îΘÇáµêÉΘçìσñìΘçìΦ┐₧
      deadSocket.onopen = null;
      deadSocket.onmessage = null;
      deadSocket.onerror = null;
      deadSocket.onclose = null;
      try {
        deadSocket.close(HEARTBEAT_TIMEOUT_CLOSE_CODE, "heartbeat timeout");
      } catch (error) {
        console.error("σà│Θù¡σ┐âΦ╖│Φ╢àµù╢Φ┐₧µÄÑσñ▒Φ┤Ñ:", error);
      }
    }

    this.heartbeatReconnectCount += 1;
    if (this.heartbeatReconnectCount > MAX_HEARTBEAT_RECONNECT_ATTEMPTS) {
      // Φ┐₧µÄÑσÅìσñìσ╗║σ╛ùΦ╡╖µ¥Ñσì┤σºïτ╗êµö╢Σ╕ìσê░ pong∩╝îσåìΘçìΦ┐₧Σ╣ƒσÅ¬µÿ»µîüτ╗¡µëôµ£ìσèíτ½»∩╝îΣ║ñτ╗Öτö¿µê╖Σ╕╗σè¿µüóσñì
      reportScene(
        ReportScene.SOCKET_CONNECT,
        JSON.stringify({
          reason: "heartbeat-reconnect-give-up",
          heartbeatReconnectCount: this.heartbeatReconnectCount,
        }),
      );
      reporterInstance.flush();
      this.updateStatus(SocketConnectionStatus.DISCONNECTED);
      return;
    }

    // σ┐àΘí╗σàêΦÉ╜τè╢µÇü∩╝Üconnect() σ£¿ CONNECTED / CONNECTING µù╢Σ╝Üτ¢┤µÄÑΦ┐öσ¢₧
    this.updateStatus(SocketConnectionStatus.CLOSE);
    // µîçµò░ΘÇÇΘü┐∩╝îΘü┐σàìµ£ìσèíτ½»σ╝éσ╕╕µ£ƒΘù┤σà¿ΘçÅσ«óµê╖τ½»Σ╗Ñσ¢║σ«ÜΦèéσÑÅσÅìσñìΘçìΦ┐₧
    this.handleReconnect(
      this.options.reconnectInterval! *
        Math.pow(2, this.heartbeatReconnectCount - 1),
    );
  }

  sendPing() {
    if (this.socket && this.socket.readyState === WebSocket.OPEN) {
      // σ£¿σÅæΘÇüσ┐âΦ╖│σëìµúÇµƒÑtokenΣ╕ÇΦç┤µÇº
      this.checkTokenConsistency();

      this.socket.send(HeartType.PING);
      this.lastPingTime = Date.now();
      authService.renewalToken();
      // Φ«╛τ╜«Φ╢àµù╢µúÇµ╡ï
      this.pingTimeout = setTimeout(() => {
        if (!this.lastPongTime || this.lastPingTime > this.lastPongTime) {
          this.missedPongCount += 1;
          reportScene(
            ReportScene.HEART_TIMEOUT,
            JSON.stringify({
              lastPingTime: this.lastPingTime,
              lastPongTime: this.lastPongTime,
              missedPongCount: this.missedPongCount,
              readyState: this.socket?.readyState ?? null,
            }),
          );
          if (this.missedPongCount >= MAX_MISSED_PONG_COUNT) {
            this.handleHeartbeatDead();
          }
        }
      }, 10000);
    }
  }

  // σÅæΘÇüµ╢êµü»∩╝êµö»µîüΦç¬σè¿ACKτí«Φ«ñ∩╝ë
  public sendMessage(message: Message): string {
    const clientId = message.clientId as string;
    // µ╖╗σèáσê░σ╛àτí«Φ«ñΘÿƒσêù(σ«₧µù╢µ╢êµü»Σ╕ìσèáσàÑΘÿƒσêù)

  }

  // σÅæΘÇüACKτí«Φ«ñ

  // σñäτÉåACKτí«Φ«ñµ╢êµü»
  private handleAck(ackMessage: Message): void {
    const clientId = ackMessage.clientId as string;
    // ack τí«Φ«ñ
    reportScene(
      ReportScene.ACK_CONFIRM,
      JSON.stringify({
        id: ackMessage.id,
        clientId,
      }),
    );

    // µ╕àΘÖñΦ«íµù╢σÖ¿
    this.clearAckTimeout(clientId);

    // Σ╗Äµ£¬τí«Φ«ñΘÿƒσêùΣ╕¡τº╗ΘÖñ
    this.messageQueue.delete(clientId);

    // ΦºªσÅæσ¢₧Φ░â
    this.options.onMessageAck?.(clientId, ackMessage);
  }

  // Φ«╛τ╜«ACKΦ╢àµù╢σñäτÉå
  private setupAckTimeout(clientId: string): void {
    if (!this.messageQueue.has(clientId)) return;

    const message = this.messageQueue.get(clientId)!;
    // µû░µ╢êµü»µ▓íµ£ë retryCount∩╝îµ¡ñσñäΦïÑσà£σ║òµêÉ maxRetries Σ╝ÜΦ«⌐Θªûµ¼íΦ╢àµù╢σ░▒τ¢┤µÄÑσêñσñ▒Φ┤Ñ∩╝îΘçìσÅæσêåµö»µ░╕Φ┐£Φ╡░Σ╕ìσê░
    const retryCount = message.retryCount || 0;

    const timer = setTimeout(() => {
      if (!this.messageQueue.has(clientId)) {
        // µ╢êµü»σ╖▓Φó½τí«Φ«ñ∩╝îµùáΘ£ÇσñäτÉå
        return;
      }
      // Φ┐₧µÄÑΣ╕ìσÅ»τö¿µù╢Σ╕ìσåìΘçìσÅæ∩╝îΣ║ñτö▒ΘçìΦ┐₧σÉÄτÜäΦíÑσü┐ΘÇ╗Φ╛æσñäτÉå∩╝îΘü┐σàìσ╛Çµ¡╗Θô╛ΘçîΘçìσñìµèòΘÇÆ
      const socketUnavailable =
        !this.socket || this.socket.readyState !== WebSocket.OPEN;
      // µúÇµƒÑµÿ»σÉªΦ╢àΦ┐çΘçìΦ»òµ¼íµò░
      if (retryCount >= this.options.maxRetries! || socketUnavailable) {
        // µ╢êµü»σÅæΘÇüσ╜╗σ║òσñ▒Φ┤Ñ
        // this.messageQueue.delete(clientId);
        this.options.onMessageFailure?.(clientId);
        return;
      }

      // µ¢┤µû░ΘçìΦ»òµ¼íµò░
      message.retryCount = (message.retryCount || 0) + 1;
      message.timestamp = Date.now();
      this.messageQueue.set(clientId, message);

      console.warn(
        `µ╢êµü»${clientId}µ£¬τí«Φ«ñ∩╝îσ░¥Φ»òΘçìσÅæ(${message.retryCount}/${this.options.maxRetries})`,
      );

      // σ░¥Φ»òΘçìσÅæµ╢êµü»
      this.send(message);

      // Θçìµû░Φ«╛τ╜«Φ╢àµù╢
      this.setupAckTimeout(clientId);
    }, this.options.ackTimeout);

    this.pendingAckTimers.set(clientId, timer);
  }

  // µ╕àΘÖñACKτ¡ëσ╛àΦ«íµù╢σÖ¿
  private clearAckTimeout(msgId: string): void {
    const timer = this.pendingAckTimers.get(msgId);
    if (timer) {
      clearTimeout(timer);
      this.pendingAckTimers.delete(msgId);
    }
  }

  // µû¡τ║┐ΘçìΦ┐₧µ£║σê╢
  public async handleReconnect(delayMs?: number) {
    if (this.isManualClose || this.isDisposed) return;

    // τ¡ëσ╛àΘçìΦ┐₧Θù┤ΘÜöµ£ƒΘù┤Σ╣ƒσ▒₧Σ║ÄΘçìΦ┐₧Φ┐çτ¿ï∩╝îΘ£Çσàêτ╜«Σ╜ì∩╝îΘü┐σàìµÅÉτñ║µ¥íΘù¬µû¡
    useMessageStore.getState().setReconnecting(true);

    const stillActive = await this.queryActiveConversation();
    // σ░¥Φ»òΘçìΦ┐₧σëì∩╝îΦïÑτö¿µê╖Σ╕ìσ£¿Σ╝ÜΦ»¥∩╝îσêÖΣ╕ìΘçìΦ┐₧
    if (stillActive === false) {
      await this.giveUpReconnectBecauseSessionEnded();
      return;
    }

    if (this.reconnectCount >= this.options.maxReconnectAttempts!) {
      console.error(
        `σ╖▓Φ╛╛σê░µ£ÇσñºΘçìΦ┐₧µ¼íµò░(${this.options.maxReconnectAttempts})∩╝îσü£µ¡óΘçìΦ┐₧`,
      );
      this.updateStatus(SocketConnectionStatus.DISCONNECTED);
      return;
    }

    this.clearReconnectTimer();
    this.reconnectCount++;

    this.reconnectTimer = setTimeout(() => {
      void this.connectIfSessionStillActive();
    }, delayMs ?? this.options.reconnectInterval);
  }

  /**
   * σÉÄσÅ░σå╗τ╗ôΣ╝ÜµèèσçáτºÆτÜäΘçìΦ┐₧σ«Üµù╢σÖ¿µÄ¿Φ┐ƒσê░σ¢₧σëìσÅ░µëìµëºΦíîπÇé
   * µÄÆσ«Üµù╢σÖ¿µù╢µƒÑσê░τÜäπÇîΦ┐ÿµ£ëΣ╝ÜΦ»¥πÇìΦ┐Öµù╢σÅ»Φâ╜σ╖▓τ╗ÅΦ┐çµ£ƒ∩╝îσ╗║Φ┐₧σëìσ┐àΘí╗σåìµƒÑΣ╕Çµ¼íπÇé
   */
  private async connectIfSessionStillActive() {
    if (this.isManualClose || this.isDisposed) return;

    const stillActive = await this.queryActiveConversation();
    if (this.isManualClose || this.isDisposed) return;
    if (stillActive === false) {
      await this.giveUpReconnectBecauseSessionEnded();
      return;
    }

    this.options.onReconnect?.(this.reconnectCount);
    console.log("reconnectTimer", this.reconnectCount);
    this.connect(true);
  }

  /** true µ£ëΣ╝ÜΦ»¥∩╝îfalse µ▓íµ£ë∩╝înull µƒÑΦ»óσñ▒Φ┤Ñ∩╝êµ▓┐τö¿σÄƒΘÇ╗Φ╛æ∩╝îΣ╗ìτä╢σ░¥Φ»òσ╗║Φ┐₧∩╝ë */
  private async queryActiveConversation(): Promise<boolean | null> {

    }
  }

    // µö╛σ╝âΘçìΦ┐₧µêûµÿÄτí«µû¡σ╝Çµù╢Σ╝ÜΦÉ╜σê░ DISCONNECTED∩╝îΣ╗ìΣ╝ÜσÉîµ¡ÑπÇé
    if (

  // σ«ëσà¿σà│Θù¡Φ┐₧µÄÑ
  public close(code?: number, reason?: string): void {
    this.isManualClose = code === MANUAL_CLOSE_CODE;
    useMessageStore.getState().setReconnecting(false);
    this.heartbeatReconnectCount = 0;
    // Φ░âτö¿µû╣σà│µÄëΦ┐₧µÄÑσÉÄσì│Σ╕óσ╝âµ£¼σ«₧Σ╛ï∩╝îσ╛àΦºªσÅæτÜäΘçìΦ┐₧Σ╕ìΦâ╜σåìµèèσ«âµïëΦ╡╖µ¥Ñσ╗║µû░Φ┐₧µÄÑ
    this.isDisposed = true;
    this.clearReconnectTimer();
    this.stopHeartbeat();
    // µ╕àΘÖñµëÇµ£ëACKΦ«íµù╢σÖ¿
    this.pendingAckTimers.forEach((timer) => clearTimeout(timer));
    this.pendingAckTimers.clear();

    if (this.socket) {
      if (this.socket.readyState === WebSocket.OPEN) {
        // σà│Θù¡Φ┐₧µÄÑ
        this.socket.close(code || MANUAL_CLOSE_CODE, reason || "τö¿µê╖Σ╕╗σè¿σà│Θù¡");
      } else {
        const pendingSocket = this.socket;
        this.socket = null;
