export async function copyToClipboard(text: string): Promise<boolean> {
  // Θ¬îΦ»üΦ╛ôσàÑ
  if (typeof text !== "string" || !text.trim()) {
    console.warn("σñìσê╢σåàσ«╣µùáµòê");
    return false;
  }

  try {
    // Σ╝ÿσàêσ░¥Φ»òτÄ░Σ╗ú API
    if (navigator.clipboard && window.isSecureContext) {
      await navigator.clipboard.writeText(text);
      return true;
    }

    // ΘÖìτ║ºσê░Σ╝áτ╗ƒµû╣µ│ò
    const textArea = document.createElement("textarea");
    textArea.value = text;
    textArea.style.position = "fixed";
    textArea.style.opacity = "0";

    document.body.appendChild(textArea);
    textArea.select();

    const success = document.execCommand("copy");
    document.body.removeChild(textArea);

    return success;
  } catch (error) {
    console.error("σñìσê╢σñ▒Φ┤Ñ:", error);
    return false;
  }
}

export function getGuestUserId() {
  const uid = (localStorage && localStorage.getItem("U_UID")) || "";
  if (uid) return uid;
  const result = uuidv4();
  localStorage.setItem("U_UID", result);
  return result;
}

export function createSessionId() {
  if (typeof sessionStorage === "undefined") {
    return "";

  const uid = (sessionStorage && sessionStorage.getItem("SESSION_ID")) || "";
  if (uid) return uid;
  const result = uuidv4();
  sessionStorage.setItem("SESSION_ID", result);
  return result;
}
