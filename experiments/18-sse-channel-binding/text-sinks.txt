const isStoreBoundIssueTypeMsg = (msg: Message): boolean => {
  if (msg.type !== MessageType.IssueType) return false;
  // Θªûσ▒Å store-bound∩╝êqr_issue_type_name∩╝ëΣ╕Äτ½ÖσåàΣ╕Çµ¼íµÇºτ¢┤Φ┐₧∩╝êoneshot key∩╝ëτÜäΘù¿σ║ùσÉìΘâ╜Φªüσî╣Θàì∩╝Ü
  // Σ╕Çµ¼íµÇºτ¢┤Φ┐₧Σ╕ìσåÖ qr_*∩╝îΘù¿σ║ùσÉìΦ╡░τï¼τ½ï oneshot key∩╝îσÉªσêÖµ£¼µ¼íΣ╝ÜΦ»¥τÜäΘù¿σ║ùτ│╗τ╗ƒµ╢êµü»µùáµ│òΘÜÉΦùÅπÇé
  const parsed = tryParse(msg.content);
  const msgText = parsed?.text ?? msg.content;
  if (typeof msgText !== "string") return false;
  const persistedName = getStoredIssueTypeCache()?.issueTypeName;
  const oneShotName = getOneShotIssueTypeName();
  return (
    (!!persistedName && msgText === persistedName) ||
    (!!oneShotName && msgText === oneShotName)
  );
};

export const formatHistory = (historyData: Message[]) => {
  return historyData.filter((item) => {
    const parseContent = tryParse(item.content);
    if (parseContent && parseContent.fileName) {
      item.content = JSON.stringify({
        ...parseContent,
        fileType: getFileType(parseContent.fileName),
      });
    }
    if (isStoreBoundIssueTypeMsg(item)) {
      return false;
    }
    return item.type !== MessageType.LineUp;
  });
};

export const formatMsgList = (msgList: Message[], newMessage: Message) => {
  if (
    isLineUpMsg(newMessage) ||
    isAgentOnline(newMessage) ||
    isChatEndMsg(newMessage) ||
    isAgentWelcome(newMessage)

      .getState()

export const getCurrentChatStatus = (newMessage: Message) => {
  if (isLineUpMsg(newMessage)) {
    return CurrentChatStatus.LineUp;
  }

  if (isAgentOnline(newMessage) || isAgentWelcome(newMessage)) {
    return CurrentChatStatus.Online;
  }

  if (isChatEndMsg(newMessage)) {
    return CurrentChatStatus.ChatEnd;
  }

  return null;
};

export const handleLineMsg = (msgList: Message[], newMessage: Message) => {
  let lineMsg = null;
  const excludeLineMsgList: Message[] = [];
  msgList.forEach((msg) => {
    if (isLineUpMsg(msg)) {
      lineMsg = msg;
    } else {
      excludeLineMsgList.push(msg);
    }
  });
  return {
    excludeLineMsgList,
    appendMsg: lineMsg ? [newMessage, lineMsg] : [newMessage],
  };
};

const handleForbiddenMsg = (message: Message, msgList: Message[]) => {
  const clientId = tryParse(message.content)?.clientId;
  if (clientId) {
    useMessageStore.getState().updateMessageStatus(clientId as string, {
      riskStatus: RiskStatus.Failed,
      status: MessageStatus.Failed,
    });
  }
  // ΦÇâΦÖæµ╢êµü»µÿ»σÉªΦó½σ╝òτö¿Σ║å∩╝îΦïÑσ¥Éσ╕¡σêÜµö╢σê░τ½ïΘ⌐¼σ╝òτö¿∩╝îµ╢êµü»Σ╣ƒΣ╝Üσ¡ÿσ£¿σ╝òτö¿µ╢êµü»σåà
  msgList.forEach((item) => {
    if (item.quoteMessage && item.quoteMessage.clientId === clientId) {
      item.quoteMessage = {
        ...item.quoteMessage,
        content: "",
        riskStatus: RiskStatus.Failed,
        type: MessageType.ForbiddenMsg,
      };
    }
  });
};

export const handleIncomingMessage = (message: Message, msgList: Message[]) => {
  message.isNewMsg = true;
  if (isForbiddenMsg(message)) {
    handleForbiddenMsg(message, msgList);
    return;
  }
  // µÄÑµö╢µÆñσ¢₧µ╢êµü»
  if (isWithdrawnMsg(message)) {
    const recallId = tryParse(message.content)?.id;
    if (!recallId) {
      return;
    }
    // µ¢┤µû░µ£¼σ£░τè╢µÇü
    useMessageStore.getState().withdrawMessage(recallId, message.senderType);

    // σêñµû¡σ╜ôσëìµ╢êµü»µÿ»σÉªµ£ëΦó½σ╝òτö¿∩╝îΦïÑΦó½σ╝òτö¿∩╝îσêÖΘ£ÇΦªüµ╕àΘÖñσåàσ«╣
    return;
  }
  if (isLineUpMsg(message)) {
    reportScene(ReportScene.LINE_UP, message.content);
  }
  if (message.type === MessageType.ConnectionSuccess) {
    message.hide = true;
    if (
      typeof window !== "undefined" &&
      window.globalData &&
      typeof window.globalData.serveSocketPreparedCallback === "function"
    ) {
      window.globalData.serveSocketPreparedCallback();
    }
  }
  if (isStoreBoundIssueTypeMsg(message)) {
    message.hide = true;
  }
  const valid = isInEnum(message.type, MessageType);
  if (!valid) {
    // σªéµ₧£Σ╕ìσ£¿µ₧ÜΣ╕╛σ«ÜΣ╣ëτÜäτ▒╗σ₧ïΦîâσ¢┤σåà∩╝îσêÖΘÜÉΦùÅ
    message.hide = true;
  }
  return message;
};

const formatContent = (content: string) => {
  // σàêΦ┐¢Φíî HTML Φ╜¼Σ╣ëΣ╗ÑΘÿ▓µ¡ó XSS
  const escapedContent = content
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#x27;");

  // σ░åµìóΦíîτ¼ªΦ╜¼µìóΣ╕║ <br> µáçτ¡╛
  return escapedContent.replace(/\n/g, "<br>");
};

export const transferSendContent = (content: string) => {
  const parseContent = tryParse(content);
  if (!parseContent || !parseContent.text) {
    return content;
  }
  parseContent.text = formatContent(parseContent.text);
  return JSON.stringify(parseContent);
};

export const getFileType = (fileName: string) => {
  const isImage = isImageFile(fileName);
  const isVideo = isVideoFile(fileName);
  const isAudio = isAudioFile(fileName);
  let type = MessageType.File;
  if (isImage || isVideo || isAudio) {
    if (isVideo) {
      type = MessageType.Video;
    }
    if (isAudio) {
      type = MessageType.Audio;
    }
    if (isImage) {
      type = MessageType.Image;
    }
  }
  return type;
};

/** ISO BMFF∩╝êMP4/M4A τ¡ë∩╝ëσëìΦïÑσ╣▓σ¡ùΦèéσåàµÿ»σÉªσç║τÄ░ M4A σà╝σ«╣σôüτëî∩╝êΘâ¿σêå m4a Σ╗àµáç isom∩╝ë */
function isoBmffContainsM4ABrand(buf: ArrayBuffer): boolean {
  const a = u8(buf);
  const max = Math.min(a.length, 16384);
  for (let i = 0; i <= max - 4; i++) {
    if (
      a[i] === 0x4d &&
      a[i + 1] === 0x34 &&
      a[i + 2] === 0x41 &&
      a[i + 3] === 0x20
    ) {
      return true;
    }
  }
  return false;
}

/**
 * Σ┐«σñìτò╕σ╜ó JSON∩╝ÜΦ╜¼Σ╣ëσ¡ùτ¼ªΣ╕▓σ¡ùΘ¥óΘçÅσåàµ£¬Φ╜¼Σ╣ëτÜäµÄºσê╢σ¡ùτ¼ª∩╝êµìóΦíî/σ¢₧Φ╜ª/σê╢Φí¿τ¼ª∩╝ë∩╝îσåìσ░¥Φ»òΦºúµ₧ÉπÇé

function tryParseRepairedJson(
  str: string,
): { text?: string; [key: string]: unknown } | undefined {
  let result = "";
  let inString = false;
  let escaped = false;
  for (const ch of str) {
    if (escaped) {
      result += ch;
      escaped = false;
      continue;
    }
    if (ch === "\\") {
      result += ch;
      escaped = true;
      continue;
    }
    if (ch === '"') {
      inString = !inString;
      result += ch;
      continue;
    }
    if (inString) {
      if (ch === "\n") {
        result += "\\n";
        continue;
      }
      if (ch === "\r") {
        result += "\\r";
        continue;
      }
      if (ch === "\t") {
        result += "\\t";
        continue;
      }
    }
    result += ch;
  }
  try {
    return JSON.parse(result);
  } catch {
    return undefined;
  }
}

export function parseMessageContent(
  str: string,

  if (!str || typeof str !== "string") return undefined;
  // µÿ»σÉªΓÇ£τ£ïΦ╡╖µ¥ÑσâÅΓÇ¥τ╗ôµ₧äσîû JSON∩╝êσ»╣Φ▒í/µò░τ╗ä∩╝ë∩╝îσå│σ«Üτò╕σ╜óσ£║µÖ»Σ╕ïµÿ»σÉªσà£σ║òµêÉτ║»µûçµ£¼
  const looksLikeJsonObject = /^\s*[{[]/.test(str);
  try {
    const parsed = JSON.parse(str);
    if (
      parsed &&
      typeof parsed === "object" &&
      typeof parsed.text === "string"
    ) {
      return parsed;
    }
    // Φºúµ₧ÉµêÉσèƒΣ╜åΣ╕ìµÿ» {text:string}
    // - τ╗ôµ₧äσîûσ»╣Φ▒í/µò░τ╗ä∩╝êσªéµûçΣ╗╢µ╢êµü» {fileName}∩╝ë∩╝ÜΣ║ñτö▒Σ╕èσ▒éσêñµû¡∩╝îµùá text Φ┐öσ¢₧ undefined
    // - σ╜óσªéτ║»µò░σ¡ùτ¡ëΘ¥₧τ╗ôµ₧äσîû∩╝Üσ╜ôΣ╜£τ║»µûçµ£¼σ▒òτñ║
    if (looksLikeJsonObject) return undefined;
    return { text: str };
  } catch {
    if (looksLikeJsonObject) {
      // τò╕σ╜ó JSON∩╝ÜσàêΣ┐«σñìµÄºσê╢σ¡ùτ¼ªσåìΦºúµ₧É
      const repaired = tryParseRepairedJson(str);
      if (repaired && typeof repaired.text === "string") return repaired;
      // µ¡úσêÖσà£σ║ò∩╝ê[\s\S] Φ╖¿Φíîσî╣Θàì∩╝ë∩╝Ü{"text":""xxx""} / {"text":"xxx"}
      const doubleQuoted = str.match(/"text"\s*:\s*""([\s\S]*?)""\s*[\},]/);
      if (doubleQuoted) return { text: doubleQuoted[1] };
      const singleQuoted = str.match(/"text"\s*:\s*"([\s\S]*?)"\s*[\},]/);
      if (singleQuoted)
        return { text: stripSurroundingQuotes(singleQuoted[1]) };

    return { text: str };
  }
}

export function removeObjectKeys<T extends object, K extends keyof T>(
  obj: T,
  keysToRemove: K[],
): Omit<T, K> {
  return Object.entries(obj).reduce(
    (acc, [key, value]) => {
      if (!keysToRemove.includes(key as K)) {
        acc[key as keyof Omit<T, K>] = value;
      }
      return acc;
    },
    {} as Omit<T, K>,
  );
}

export function getAllUrlParams(): Record<string, string> {
  const params = getUrlParams();
  const result: Record<string, string> = {};

  params.forEach((value, key) => {
    result[key] = value;
  });

  return result;
}

export const isValidUrl = (url: string): boolean => {
  try {
    if (!url) return false;
    const parsedUrl = new URL(url);
    // Only allow http and https protocols
    return ["http:", "https:"].includes(parsedUrl.protocol.toLowerCase());
  } catch (e) {
    // If URL parsing fails, it's likely not a valid URL
    return false;
  }
};
