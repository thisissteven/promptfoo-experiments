import request from "@/request";

export interface MallCardSnapshotRequest {
  cardType: "product" | "order";
  site?: string;
  orderNo?: string;
  productId?: string;
  businessId?: string;
  skuId?: string;
  product?: string;
  clientId?: string;
  requestId?: string;
  aiUserId?: string;
  aiConversationId?: string;

export const createMallCardSnapshot = async (
  data: MallCardSnapshotRequest,
): Promise<MallContextCard> => {
  const result = await request.post<MallContextCard>(
    "/gw/robot/im/mall/card/snapshot",
    { ...data },
    { timeout: 20000, notToastError: true },
  );
  const card = parseMallContextCard(result);
  if (!card) {
    throw new Error("Invalid mall card snapshot response");
  }
  return card;
};

import { createMallCardSnapshot } from "@/app/api/mallCard";
import { authService } from "@/app/services/AuthService";
import { useMessageStore } from "@/app/store";
import {
  HeaderType,
  MessageSender,
  MessageType,
  RobotMessageType,
} from "@/app/store/types";

const MALL_CARD_QUERY_KEYS = ["contextType", "businessId", "product"] as const;

interface MallCardQuery {
  cardType: "product" | "order";
  businessId?: string;
  orderNo?: string;
  product?: string;
}

export const getMallWsContextFromUrl = (
  searchParams?: URLSearchParams,
): {
  contextType: "product" | "order";
  businessId: string;
  product?: string;
} | null => {
  if (typeof window === "undefined") return null;
  const params =
    searchParams ?? new URLSearchParams(window.location.search || "");
  const contextType = params.get("contextType")?.trim().toUpperCase();
  const businessId = params.get("businessId")?.trim();
  const product = params.get("product")?.trim() || undefined;
  if ((contextType !== "PRODUCT" && contextType !== "ORDER") || !businessId) {
    return null;
  }
  return {
    contextType: contextType === "ORDER" ? "order" : "product",
    businessId,
    ...(product && { product }),
  };
};

const readQuery = (searchParams: URLSearchParams): MallCardQuery | null => {
  const contextType = searchParams.get("contextType")?.trim().toUpperCase();
  if (contextType !== "PRODUCT" && contextType !== "ORDER") {
    return null;
  }

  const businessId = searchParams.get("businessId")?.trim() || undefined;
  const product = searchParams.get("product")?.trim() || undefined;
  if (!businessId) return null;
  if (contextType === "ORDER") {
    return {
      cardType: "order",
      businessId,
      ...(product && { product }),
    };
  }
  return {
    cardType: "product",
    businessId,
    ...(product && { product }),
  };
};

export const clearMallContextQuery = () => {
  if (typeof window === "undefined") return;
  const url = new URL(window.location.href);
  let changed = false;
  MALL_CARD_QUERY_KEYS.forEach((key) => {
    if (url.searchParams.has(key)) {
      url.searchParams.delete(key);
      changed = true;
    }
  });
  if (changed) {
    window.history.replaceState({}, document.title, url.toString());
  }
};

const getClientId = (query: MallCardQuery) => {
  const storageKey = `mall_card_snapshot_client_id:${JSON.stringify(query)}`;
  const stored = sessionStorage.getItem(storageKey);
  if (stored) return { clientId: stored, storageKey };

  const clientId = `mall-${query.cardType}-${createMessageClientId()}`;
  sessionStorage.setItem(storageKey, clientId);
  return { clientId, storageKey };
};

const hasRenderedCard = (clientId: string) => {
  const { historyMsgList, msgList } = useMessageStore.getState();
  return [...historyMsgList, ...msgList].some(
    (item) => item.clientId === clientId,
  );
};

export const renderMallCardFromUrl = async (

): Promise<void> => {

  if (!authService.getToken()) return;

    searchParams ?? new URLSearchParams(window.location.search || "");

        clientId: responseClientId,
        type:
          card.cardType === "product"

  }
